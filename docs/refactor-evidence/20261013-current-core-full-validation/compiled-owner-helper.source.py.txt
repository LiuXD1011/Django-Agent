import datetime

ROOT = Path('/home/liuxuedeng/Django-Agent')

def write(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def frozen():
    original = json.loads((OUT / 'source_inputs.before.json').read_text())
    return {name: sha(ROOT / name) for name in original}

def deps():
    original = json.loads((OUT / 'installed-dependency-inputs.json').read_text())
    return {name: sha(CORE / name) for name in original}

def inventory():
    return {str(p.relative_to(CORE)): {'sha256': sha(p), 'size': p.stat().st_size, 'mtime_ns': p.stat().st_mtime_ns} for p in sorted((CORE / 'dist').rglob('*')) if p.is_file()}

def observed(pid):
    parents = {}
    for proc in Path('/proc').iterdir():
        if proc.name.isdigit():
            try:
                status = (proc / 'status').read_text()
                parents[int(proc.name)] = int(re.search(r'^PPid:\s+(\d+)$', status, re.M)[1])
            except (OSError, TypeError):
                pass
    children = {pid}
    while True:
        expanded = children | {p for p, parent in parents.items() if parent in children}
        if expanded == children:
            break
        children = expanded
    result = {}
    for p in children:
        proc = Path('/proc') / str(p)
        try:
            args = (proc / 'cmdline').read_bytes().decode().split('\0')[:-1]
            executable = os.readlink(proc / 'exe')
            if 'python' in executable:
                safe_args = [v if i < 4 or v.endswith('/worker.py') else '<supervisor-source-omitted>' for i, v in enumerate(args)]
            else:
                safe_args = args
            threads = sorted(set(t.joinpath('comm').read_text().strip() for t in (proc / 'task').iterdir()))
            result[str(p)] = {'ppid': parents.get(p), 'executable': executable, 'argv': safe_args, 'thread_names': threads}
        except (OSError, UnicodeError):
            pass
    return result

def run(name, command, env, observe=False):
    start = time.time()
    sightings = {}
    with (OUT / (name + '.stdout.log')).open('wb') as stdout, (OUT / (name + '.stderr.log')).open('wb') as stderr:
        child = subprocess.Popen(command, cwd=CORE, env=env, stdout=stdout, stderr=stderr)
        if observe:
            while child.poll() is None:
                for pid, value in observed(child.pid).items():
                    if pid in sightings:
                        value['thread_names'] = sorted(set(value['thread_names']) | set(sightings[pid]['thread_names']))
                    sightings[pid] = value
                time.sleep(0.025)
        code = child.wait()
    record = {'command': command, 'cwd': str(CORE), 'pid': child.pid, 'started_utc': datetime.datetime.fromtimestamp(start, datetime.timezone.utc).isoformat(), 'elapsed_seconds': time.time() - start, 'exit_code': code, 'stdout': name + '.stdout.log', 'stderr': name + '.stderr.log'}
    write(name + '.result.json', record)
    if observe:
        write(name + '.owned-process-observation.json', sightings)
    print(json.dumps(record), flush=True)
    return code
