const hash = value => createHash('sha256').update(value).digest('hex');
const canonical = value => typeof value === 'bigint' ? JSON.stringify(value.toString()) : Array.isArray(value)
  ? `[${value.map(canonical).join(',')}]` : value && typeof value === 'object'
  ? `{${Object.entries(value).sort(([a], [b]) => a.localeCompare(b)).map(([key, item]) => `${JSON.stringify(key)}:${canonical(item)}`).join(',')}}` : JSON.stringify(value);

const fail = code => { throw Object.assign(new Error(code), { code }); };

function rootPath(path) {
  expect(typeof path === 'string' && resolve(path) === path, 'absolute_root_required');
  let cursor = '/';
  for (const part of path.slice(1).split('/')) { cursor = join(cursor, part); if (existsSync(cursor)) expect(!lstatSync(cursor).isSymbolicLink(), 'root_symlink_refused'); }
  if (existsSync(path)) expect(realpathSync(path) === path && lstatSync(path).isDirectory(), 'root_invalid');
  return path;
}
function portable(path) {
  expect(typeof path === 'string' && path.length > 0 && !/[\\:\x00-\x1f]/.test(path) && !path.startsWith('/'), 'unsafe_relative_path');
  for (const part of path.split('/')) expect(part && part !== '.' && part !== '..' && !/[<>"|?*]/.test(part)
    && !/[. ]$/.test(part) && !/^(?:con|prn|aux|nul|com[1-9]|lpt[1-9])(?:\.|$)/i.test(part), 'nonportable_path');
  return path;
}
function same(left, right, code) { expect(digest(left) === digest(right), code); }
async function inventory(root) {
  rootPath(root); const api = await runtime(), directories = [''], files = [], seen = new Set();
  const walk = path => {
    for (const name of readdirSync(join(root, path)).sort()) {
      const next = path ? path + '/' + name : name; portable(next);
      const folded = next.normalize('NFC').toLowerCase(); expect(!seen.has(folded), 'path_collision'); seen.add(folded);
      const full = join(root, next), stat = lstatSync(full); expect(!stat.isSymbolicLink(), 'file_symlink_refused');
      expect(!/-wal$|-shm$/.test(name), 'source_sidecar_refused');
      if (stat.isDirectory()) { directories.push(next); walk(next); }
      else { expect(stat.isFile(), 'special_file_refused'); files.push({ path: next, size: stat.size, sha256: api.fileHash(full) }); }
    }
  };
  walk(''); return { directories: directories.sort(), files: files.sort((a, b) => a.path.localeCompare(b.path)) };
}
function runtimeFiles() {
  const files = {};
  const walk = path => { for (const name of readdirSync(path).sort()) { const child = join(path, name); if (lstatSync(child).isDirectory()) walk(child);
    else { expect(lstatSync(child).isFile() && !lstatSync(child).isSymbolicLink(), 'runtime_file_invalid'); files[relative(PROJECT, child)] = hash(readFileSync(child)); } } };
  walk(join(CORE, 'dist'));
  for (const path of ['package.json', 'package-lock.json', 'node_modules/better-sqlite3/build/Release/better_sqlite3.node', 'node_modules/sqlite-vec/index.mjs']) {
    const full = join(CORE, path); expect(existsSync(full), 'runtime_dependency_missing'); files[relative(PROJECT, full)] = hash(readFileSync(full));
  }
  return files;
}
function readBound(path, sha256) {
  expect(typeof path === 'string' && /^[a-f0-9]{64}$/.test(sha256 || ''), 'binding_identity_required');
  const bytes = readFileSync(path); expect(hash(bytes) === sha256, 'binding_sha_mismatch'); return JSON.parse(bytes.toString('utf8'));
}
function stopped(receipt) {
  expect(receipt.protocol === 1 && receipt.kind === 'marlin-offline-source-stop' && receipt.source_root === SOURCE, 'stop_receipt_scope');
  expect(receipt.normal_stop === true && receipt.all_source_writers_exited === true && receipt.other_writers === 0, 'source_writer_unknown');
  for (const name of ['active_jobs', 'active_tasks', 'active_actors', 'pending_ipc_receipts']) expect(receipt[name] === 0, 'source_activity_refused');
  expect(Array.isArray(receipt.exited_owner_identities) && receipt.exited_owner_identities.length > 0, 'owner_identity_required');
  for (const owner of receipt.exited_owner_identities) {
    expect(Number.isSafeInteger(owner.pid) && owner.pid > 0 && /^\d+$/.test(owner.start_ticks || '') && owner.exit_code === 0, 'owner_identity_invalid');
    const stat = `/proc/${owner.pid}/stat`;
    if (existsSync(stat)) { const text = readFileSync(stat, 'utf8'), parts = text.slice(text.lastIndexOf(')') + 2).split(/\s+/);
      expect(parts[19] !== owner.start_ticks, 'same_owner_still_exists'); }
  }
}
async function databaseState(path) {
  const api = await runtime(), db = api.openNativeDatabase(path, { readonly: true, fileMustExist: true });
  try {
    db.defaultSafeIntegers(); await api.loadVec(db); db.pragma('query_only=ON');
    expect(db.pragma('integrity_check').length === 1 && db.pragma('integrity_check')[0].integrity_check === 'ok', 'database_integrity_failed');
    expect(db.pragma('foreign_key_check').length === 0, 'database_foreign_keys_failed');
    const schema = db.prepare("SELECT type,name,tbl_name,sql FROM sqlite_master WHERE name NOT LIKE 'sqlite_%' ORDER BY type,name").all();
    const tables = {};
    for (const row of schema.filter(row => row.type === 'table')) {
      expect(/^[A-Za-z_][A-Za-z_0-9]*$/.test(row.name), 'table_name_invalid');
      const rows = db.prepare(`SELECT * FROM "${row.name}"`).all(); tables[row.name] = { count: rows.length, digest: digest(rows.map(digest).sort()) };
    }
    return { schema_sha256: digest(schema), tables };
  } finally { db.close(); }
}
function idleAndMedia(db, files, api) {
  for (const [table, states] of [['desktop_jobs', ['queued', 'running']], ['task_records', ['pending', 'running']], ['agent_actors', ['queued', 'running', 'pending']]])
    expect(Number(db.get(`SELECT count(*) AS n FROM "${table}" WHERE status IN (${states.map(() => '?').join(',')})`, ...states).n) === 0, 'snapshot_activity_refused');
  const byPath = new Map(files.map(item => [item.path, item]));
  // The formal adapter persists this journal only while confirmation is pending.
  expect(!db.get("SELECT 1 AS present FROM desktop_meta WHERE key='ipc.receipts'"), 'snapshot_ipc_receipts_refused');
  expect(!db.get("SELECT 1 AS present FROM generic_resources WHERE json_extract(data,'$.generation.status')='running' LIMIT 1"), 'snapshot_generation_refused');
  let verified = 0, external = 0;
  const check = (path, expectedHash) => {
    if (/^(?:https?:|data:)/i.test(path)) { external++; return; }
    portable(path); const file = byPath.get(path); expect(file && /^[a-f0-9]{64}$/i.test(expectedHash || '') && file.sha256 === expectedHash.toLowerCase(), 'managed_media_unverified'); verified++;
  };
  for (const row of db.all("SELECT file_path,file_hash FROM knowledges WHERE type='file' AND deleted_at IS NULL AND file_path<>''")) check(String(row.file_path), String(row.file_hash));
  for (const row of db.all("SELECT storage_path,content_hash FROM knowledge_images WHERE deleted_at IS NULL AND storage_path<>''")) check(String(row.storage_path), String(row.content_hash));
  return { verified_local_references: verified, external_references_not_verified: external };
}
async function exportCopy(source, target, expected) {
  const api = await runtime(); api.assertSeparateRoots(source, target); rootPath(source); rootPath(target);
  same(await inventory(source), expected, 'source_baseline_mismatch');
  const paths = expected.files.filter(file => file.path === 'marlin.sqlite3' || SCOPE.some(dir => file.path.startsWith(dir + '/')));
  expect(paths.some(file => file.path === 'marlin.sqlite3'), 'source_database_missing');
  expect(!paths.some(file => file.path.endsWith('.lock')), 'scoped_lock_file_refused');
  const stage = new api.StagedDirectory(target, source); let db;
  try {
    const raw = join(stage.path, 'raw'), snapshot = join(stage.path, 'snapshot'); mkdirSync(raw); mkdirSync(snapshot);
    // Original root: only the audited byte-copy helper; no native DB constructors.
    api.copyVerifiedFiles(source, raw, paths); same(await inventory(source), expected, 'source_changed_after_byte_copy');
    const original = await databaseState(join(raw, 'marlin.sqlite3'));
    const nested = new Map();
    for (const file of paths.filter(file => file.path !== 'marlin.sqlite3' && /\.sqlite3?$|\.db$/i.test(file.path))) nested.set(file.path, await databaseState(join(raw, file.path)));
    const ro = api.openNativeDatabase(join(raw, 'marlin.sqlite3'), { readonly: true, fileMustExist: true });
    try { await ro.backup(join(snapshot, 'marlin.sqlite3')); } finally { ro.close(); }
    const copied = await api.copyDataFiles(raw, snapshot);
    same(copied.map(file => file.path).sort(), paths.filter(file => file.path !== 'marlin.sqlite3').map(file => file.path).sort(), 'snapshot_scope_changed');
    for (const file of copied) { const prior = paths.find(item => item.path === file.path); expect(prior, 'unexpected_copy_file');
      if (nested.has(file.path)) same(await databaseState(join(snapshot, file.path)), nested.get(file.path), 'nested_database_changed');
      else expect(file.size === prior.size && file.sha256 === prior.sha256, 'ordinary_file_changed'); }
    same(await databaseState(join(snapshot, 'marlin.sqlite3')), original, 'snapshot_database_changed');
    db = new api.CoreDatabase(snapshot);
    const media = idleAndMedia(db, copied, api);
    same(await databaseState(join(snapshot, 'marlin.sqlite3')), original, 'snapshot_initialization_changed');
    const manifest = await api.createBackup(db, join(stage.path, 'backup')); db.close(); db = undefined;
    same(manifest.tables, original.tables, 'formal_backup_tables_changed');
    same(manifest.files.map(file => file.path).sort(), paths.map(file => file.path).sort(), 'formal_backup_scope_changed');
    for (const file of manifest.files.filter(file => file.path !== 'marlin.sqlite3')) {
      if (nested.has(file.path)) same(await databaseState(join(stage.path, 'backup', file.path)), nested.get(file.path), 'formal_nested_database_changed');
      else same(file, paths.find(prior => prior.path === file.path), 'formal_ordinary_file_changed');
    }
    same(await inventory(source), expected, 'source_changed_after_backup');
    api.writePrivateJson(join(stage.path, 'offline-clone-manifest.json'), { protocol: 1, kind: 'marlin-offline-clone',
      source_tree_sha256: digest(expected), source_files: expected.files.length, copied_files: paths.length, ordinary_file_byte_hash_verified: true,
      native_backup_only_on_owned_raw: true, nested_sqlite_logical_verified: [...nested.keys()], ...media,
      backup_manifest_sha256: api.fileHash(join(stage.path, 'backup/backup-manifest.json')) });
    // Raw native reader sidecars are owned here; only the verified formal backup is published.
    rmSync(raw, { recursive: true }); rmSync(snapshot, { recursive: true }); stage.publish();
    return { protocol: 1, source_unchanged: true, source_files: expected.files.length, backup_files: manifest.files.length,
      backup_tables: Object.keys(manifest.tables).length, nested_sqlite_databases: nested.size, ...media,
      manifest_sha256: api.fileHash(join(target, 'offline-clone-manifest.json')) };
  } finally { db?.close(); stage.finish(); }
}

async function main() {
  const { values } = parseArgs({ options: { phase: { type: 'string', default: 'review_only' }, binding: { type: 'string' }, 'binding-sha': { type: 'string' } }, strict: true });
  if (values.phase === 'review_only') return { ok: true, phase: values.phase, source_root: SOURCE, target_root: TARGET,
    script_sha256: hash(readFileSync(scriptPath)), runtime_files: runtimeFiles(), executes_copy: false, actual_data_read: false };
  if (values.phase === 'prepare-test') return prepareTests();
  expect(values.phase === 'export', 'unknown_phase');
  const request = readBound(values.binding, values['binding-sha']);
  expect(request.protocol === 1 && request.operation === 'marlin-offline-clone-export' && request.source_root === SOURCE && request.target_root === TARGET, 'request_scope_refused');
  expect(request.script_sha256 === hash(readFileSync(scriptPath)), 'harness_sha_mismatch'); same(runtimeFiles(), request.runtime_files, 'runtime_binding_mismatch');
  stopped(readBound(request.stop_receipt.path, request.stop_receipt.sha256));
  const list = readBound(request.source_inventory.path, request.source_inventory.sha256);
  expect(list.protocol === 1 && list.source_root === SOURCE && list.offline_after_stop === true && Array.isArray(list.files) && Array.isArray(list.directories), 'inventory_binding_invalid');
  const expected = { files: list.files, directories: list.directories }; same(await inventory(SOURCE), expected, 'original_inventory_mismatch');
  return { ok: true, phase: 'export', ...await exportCopy(SOURCE, TARGET, expected), root_binding_sha256: values['binding-sha'] };
}
main().then(value => process.stdout.write(JSON.stringify(value, null, 2) + '\n')).catch(error => {
  process.stderr.write(JSON.stringify({ ok: false, code: typeof error.code === 'string' ? error.code : 'offline_clone_validation_failed', secret_values_reported: false }) + '\n'); process.exitCode = 1;
});
