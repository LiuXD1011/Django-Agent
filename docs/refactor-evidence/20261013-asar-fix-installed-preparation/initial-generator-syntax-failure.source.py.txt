def walk_regular(root):
    safe_path(root)
    need(Path(root).is_dir(), 'artifact_directory_missing')
    def visit(folder):
        with os.scandir(folder) as entries:
            for entry in sorted(entries, key=lambda value: value.name):
                file = Path(entry.path)
                info = entry.stat(follow_symlinks=False)
                need(not stat.S_ISLNK(info.st_mode) and not getattr(info, 'st_file_attributes', 0) & 1024, 'artifact_reparse_rejected')
                if stat.S_ISDIR(info.st_mode):
                    yield from visit(file)
                else:
                    need(stat.S_ISREG(info.st_mode), 'artifact_nonregular_file')
                    yield file
    yield from visit(Path(root))
