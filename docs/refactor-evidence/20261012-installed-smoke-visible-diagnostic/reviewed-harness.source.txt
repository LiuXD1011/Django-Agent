def user32_api():

    u = ctypes.WinDLL("user32", use_last_error=True)
    callback = ctypes.WINFUNCTYPE(w.BOOL, w.HWND, w.LPARAM)

    need(ctypes.sizeof(Rect) == 16 and ctypes.sizeof(w.DWORD) == 4, "win32_rect_or_dword_layout_invalid")
    signatures = {
      "EnumWindows": ([callback, w.LPARAM], w.BOOL),
      "IsWindow": ([w.HWND], w.BOOL),
      "IsWindowVisible": ([w.HWND], w.BOOL),
      "GetWindowThreadProcessId": ([w.HWND, ctypes.POINTER(w.DWORD)], w.DWORD),
      "GetClientRect": ([w.HWND, ctypes.POINTER(Rect)], w.BOOL),
      "GetWindow": ([w.HWND, w.UINT], w.HWND),
      "GetClassNameW": ([w.HWND, w.LPWSTR, ctypes.c_int], ctypes.c_int),
      "ShowWindow": ([w.HWND, ctypes.c_int], w.BOOL),
      "GetForegroundWindow": ([], w.HWND),
    }
    for name, (arguments, result) in signatures.items(): getattr(u, name).argtypes, getattr(u, name).restype = arguments, result
    return u, w, Rect, callback

def owned_window(api, hwnd, pid):
    u, w, Rect, _ = api
    if not u.IsWindow(hwnd): return None
    owner = w.DWORD(); need(u.GetWindowThreadProcessId(hwnd, ctypes.byref(owner)), "window_pid_unavailable")
    if owner.value != pid: return None
    rectangle = Rect(); need(u.GetClientRect(hwnd, ctypes.byref(rectangle)), "owned_client_rect_unavailable")
    name = ctypes.create_unicode_buffer(256); need(u.GetClassNameW(hwnd, name, len(name)), "owned_window_class_unavailable")
    width, height = rectangle.right - rectangle.left, rectangle.bottom - rectangle.top
    if name.value != "Chrome_WidgetWin_1" or u.GetWindow(hwnd, 4) or not (700 <= width <= 4096 and 500 <= height <= 2160): return None
    return {"hwnd": int(hwnd), "pid": int(owner.value), "class_name": name.value, "client_width": width, "client_height": height, "visible": bool(u.IsWindowVisible(hwnd))}
