// Only authenticated transport and unrelated settings children are isolated.
const directory = fileURLToPath(new URL('.', import.meta.url))
const parent = new URL('../Settings.vue', import.meta.url)
const child = new URL('./WebSearchSettings.vue', import.meta.url)
const compile = (path, id) => {
  const { descriptor } = parse(readFileSync(path, 'utf8'))
  return compileScript(descriptor, { id, inlineTemplate: true }).content.replace('export default', 'const Component=')
    + `\nComponent.__scopeId='data-v-${id}';export default Component;`
}
let bundlePromise
function bundle() {
  return bundlePromise ??= build({
    stdin: { contents: `import Settings from 'actual:settings';import {createApp,h} from 'vue';import TDesign from 'tdesign-vue-next';
      import {createRouter,createMemoryHistory,RouterView} from 'vue-router';
      const router=createRouter({history:createMemoryHistory(),routes:[{path:'/platform/settings',component:Settings}]});
      const app=createApp({render:()=>h(RouterView)});app.use(router).use(TDesign);window.fixtureApp=app;
      router.push(location.pathname+location.search).then(()=>router.isReady()).then(()=>app.mount('#app'));`,
      loader: 'ts', resolveDir: directory },
    bundle: true, write: false, format: 'iife', loader: { '.css': 'empty' },
    plugins: [{ name: 'settings-provider-boundaries', setup(builder) {
      builder.onResolve({ filter: /^actual:/ }, args => ({ path: args.path, namespace: 'actual' }))
      builder.onResolve({ filter: /WebSearchSettings\.vue$/ }, () => ({ path: 'actual:search', namespace: 'actual' }))
      builder.onLoad({ filter: /.*/, namespace: 'actual' }, args => ({ contents: compile(args.path === 'actual:settings' ? parent : child,
        args.path === 'actual:settings' ? 'settings-provider-parent' : 'settings-provider-search'), loader: 'ts',
        resolveDir: args.path === 'actual:settings' ? fileURLToPath(new URL('..', import.meta.url)) : directory }))
      builder.onResolve({ filter: /stores\/auth$|\.vue$/ }, args => ({ path: args.path, namespace: 'unrelated' }))
      builder.onLoad({ filter: /.*/, namespace: 'unrelated' }, args => ({ contents: args.path.endsWith('/auth')
        ? `export const useAuthStore=()=>({});` : `export default {render(){return null}};`, loader: 'js' }))
      builder.onResolve({ filter: /^\.\/(client|auth-transport\.mjs|endpoint\.mjs)$/ }, args => (process.stdout.write(JSON.stringify({diagnostic:'esbuild_transport_importer',importer:args.importer,matched:args.importer.endsWith('/api/index.ts')})+'\n'),args.importer.endsWith('/api/index.ts'))
        ? { path: args.path, namespace: 'transport' } : undefined)
      builder.onLoad({ filter: /.*/, namespace: 'transport' }, args => ({ contents: args.path === './client'
        ? `const request=async(method,path,body)=>{const response=await fetch(path,{method,headers:{'content-type':'application/json'},...(body===undefined?{}:{body:JSON.stringify(body)})});
          const value=await response.json();if(!response.ok)throw{...value,status:response.status};return value};
          export default{get:p=>request('GET',p),post:(p,b)=>request('POST',p,b),put:(p,b)=>request('PUT',p,b),delete:p=>request('DELETE',p)};`
        : args.path === './auth-transport.mjs' ? `export const authTransport={};` : `export const endpoint={};`, loader: 'js' }))
    } }],
  }).then(result => result.outputFiles[0].text)
}
function styles() {
  const result = [readFileSync(new URL('../../styles/app.css', import.meta.url), 'utf8'),
    readFileSync(new URL('../../styles/mobile.css', import.meta.url), 'utf8'),
    readFileSync(new URL('../../../node_modules/tdesign-vue-next/es/style/index.css', import.meta.url), 'utf8')]
  for (const [path, id] of [[parent, 'settings-provider-parent'], [child, 'settings-provider-search']]) {
    if (!existsSync(path)) continue
    for (const style of parse(readFileSync(path, 'utf8')).descriptor.styles) {
      const compiled = compileStyle({ source: style.content, filename: fileURLToPath(path), id: `data-v-${id}`, scoped: style.scoped })
      assert.deepEqual(compiled.errors, []); result.push(compiled.code)
    }
  }
  return result.join('\n')
}
const provider = (patch = {}) => ({ id: 'search-a', tenant_id: 2, name: '本地搜索', provider: 'searxng', url: 'http://127.0.0.1:44702/search',
  enabled: true, status: 'active', cx: '', description: '', created_at: '2026-10-12T00:00Z', updated_at: '2026-10-12T00:00Z',
  credential_fields: { api_key: { configured: true } }, credentials_configured: true, ...patch })
async function fixture(options = {}) {
  const browser = await chromium.launch({ headless: true })
  const page = await browser.newPage({ viewport: { width: 1280, height: 860 } })
  const state = { providers: options.providers || [], config: options.config || { enabled: false, notes: '保留备注' }, requests: [], errors: [],
    mcp: options.mcp || [], nextId: 1 }
  page.on('pageerror', error => state.errors.push(error.message))
  await page.route('**/*', async route => {
    const request = route.request(), path = new URL(request.url()).pathname, method = request.method()
    if (path === '/platform/settings') return route.fulfill({ contentType: 'text/html', body: '<div class="shell"><aside class="sidebar"></aside><section class="workspace"><div id="app"></div></section></div>' })
    const body = request.postData() ? JSON.parse(request.postData()) : undefined
    state.requests.push({ method, path, body })
    const override = await options.respond?.({ method, path, body }, state)
    if (override) return route.fulfill({ status: override.status || 200, contentType: 'application/json', body: JSON.stringify(override.body) })
    let data = {}
    if (path === '/api/v1/auth/me') data = { user: { id: 'alice', username: 'fixture', is_system_admin: false, tenant_id: '2' },
      tenant: { id: '2', name: 'fixture', space_type: options.role === 'viewer' ? 'organization' : 'personal', personal_owner_id: 'alice' }, memberships: [{ tenant_id: '2', role: options.role || 'owner' }] }
    else if (path.endsWith('/types')) data = { items: ['duckduckgo', 'searxng', 'google', 'bing'].map(provider => ({ provider })) }
    else if (path === '/api/v1/web-search-providers' && method === 'GET') data = { items: state.providers, total: state.providers.length }
    else if (path === '/api/v1/web-search-providers' && method === 'POST') {
      const saved = provider({ ...body, id: 'search-' + state.nextId++, credential_fields: { api_key: { configured: !!body.api_key } }, credentials_configured: !!body.api_key }); delete saved.api_key
      state.providers.push(saved); data = saved
    } else if (path === '/api/v1/tenants/kv/web-search-config') {
      if (method === 'PUT') state.config = { ...state.config, ...body.value }
      data = { value: state.config }
    } else if (path.startsWith('/api/v1/tenants/kv/')) data = { value: {} }
    else if (path === '/api/v1/mcp-services') data = { items: state.mcp }
    else if (path.startsWith('/api/v1/mcp-services/') && method === 'PUT') { state.mcp[0] = { ...state.mcp[0], ...body }; delete state.mcp[0].api_key; data = state.mcp[0] }
    else if (path.startsWith('/api/v1/web-search-providers/')) {
      const id = decodeURIComponent(path.split('/')[4]), row = state.providers.find(item => item.id === id)
      if (path.endsWith('/test')) data = { state: 'connected', result_count: 0 }
      else if (path.includes('/credentials/')) { const field = decodeURIComponent(path.split('/').at(-1)); row.credential_fields[field] = { configured: false }; row.credentials_configured = Object.values(row.credential_fields).some(value => value.configured); data = { fields: row.credential_fields } }
      else if (method === 'PUT') { Object.assign(row, body); if (body.api_key) { row.credential_fields.api_key = { configured: true }; row.credentials_configured = true; delete row.api_key }; data = row }
      else if (method === 'DELETE') { state.providers = state.providers.filter(item => item.id !== id); data = {} }
    }
    return route.fulfill({ contentType: 'application/json', body: JSON.stringify({ success: true, data }) })
  })
  try {
    await page.goto('http://settings.fixture.test/platform/settings?section=' + (options.section || 'websearch'))
    await page.addStyleTag({ content: styles() })
    const identity = page.waitForResponse(response => new URL(response.url()).pathname === '/api/v1/auth/me')
    await page.addScriptTag({ content: await bundle() })
    await identity
    await page.getByRole('heading', { name: '设置', exact: true }).waitFor()
  } catch (error) { await browser.close(); throw error }
  return { page, state, close: () => browser.close() }
}

test('SEARCH-UI-08 settings and the actual editor fit supported screen widths', async () => {
  const f = await fixture({ providers: [provider({ name: '较长的搜索服务名称与用途说明', url: 'http://localhost:8080/search/with/a/long/path' })] })
  try {
    for (const [width, height] of [[1440, 900], [1024, 768], [390, 844]]) {
      await f.page.setViewportSize({ width, height })
      await f.page.getByRole('button', { name: '添加搜索服务', exact: true }).click()
      const dialog = f.page.locator('.t-dialog').filter({ hasText: '添加搜索服务' })
      await dialog.waitFor()
      const dimensions = await f.page.evaluate(() => ({ viewport: innerWidth, scroll: document.documentElement.scrollWidth,
        controls: [...document.querySelectorAll('.search-editor input,.search-editor select,.search-editor textarea,.search-editor button')].map(node => {
          const rect = node.getBoundingClientRect(); return { left: rect.left, right: rect.right, width: rect.width }
        }) }))
      assert.ok(dimensions.scroll <= dimensions.viewport + 1, `${width}: page overflow`)
      for (const control of dimensions.controls) assert.ok(control.width > 0 && control.left >= -1 && control.right <= width + 1, `${width}: clipped editor ${JSON.stringify(control)}`)
      await dialog.getByRole('button', { name: '取消', exact: true }).click(); await dialog.waitFor({ state: 'hidden' })
    }
    assert.deepEqual(f.state.errors, [])
  } finally { console.log(JSON.stringify({diagnostic:'fixture_requests',requests:f.state.requests.map(({method,path,body})=>({method,path,bodyKeys:Object.keys(body||{})})),alerts:await f.page.locator('[role=alert]').allTextContents()}));await f.close() }
})

test('MCP-UI-EDIT-02 cancellation clears the hidden password input and new keys submit exactly once', async () => {
  const f = await fixture({ section: 'mcp', mcp: [{ id: 'mcp-a', name: '受保护服务', description: '用途', url: 'http://localhost:8080/mcp', enabled: true,
    credential_fields: { api_key: { configured: true } } }] })
  try {
    await f.page.getByRole('button', { name: '编辑', exact: true }).click()
    const dialog = f.page.locator('.t-dialog').filter({ hasText: '编辑 MCP 服务' }), key = dialog.locator('input[type=password]')
    await key.fill('unsubmitted-mcp-key'); await dialog.getByRole('button', { name: '取消', exact: true }).click()
    await dialog.waitFor({ state: 'hidden' })
    assert.equal(await f.page.locator('input[type=password]').evaluateAll(nodes => nodes.some(node => node.value === 'unsubmitted-mcp-key')), false)
    assert.equal(f.state.requests.some(request => request.method === 'PUT'), false)
    await f.page.getByRole('button', { name: '编辑', exact: true }).click()
    assert.equal(await key.inputValue(), '')
    await key.fill('synthetic-new-mcp-key'); await dialog.getByRole('button', { name: '保存', exact: true }).click()
    await f.page.getByText('MCP 服务已保存', { exact: true }).waitFor()
    assert.equal(await f.page.locator('input[type=password]').evaluateAll(nodes => nodes.some(node => node.value === 'synthetic-new-mcp-key')), false)
    const writes = f.state.requests.filter(request => request.method === 'PUT')
    assert.equal(writes.length, 1); assert.equal(writes[0].body.api_key, 'synthetic-new-mcp-key')
    assert.deepEqual(Object.keys(writes[0].body).sort(), ['api_key', 'description', 'enabled', 'name', 'url'])
    assert.deepEqual(f.state.errors, [])
  } finally { console.log(JSON.stringify({diagnostic:'fixture_requests',requests:f.state.requests.map(({method,path,body})=>({method,path,bodyKeys:Object.keys(body||{})})),alerts:await f.page.locator('[role=alert]').allTextContents()}));await f.close() }
})
