def write_new(path, value):
    with path.open('x') as stream:
        stream.write(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def digest(rows):
    hashes = []
    for row in rows:
        value = {name: {'bytes': len(item), 'sha256': hashlib.sha256(item).hexdigest()} if isinstance(item, bytes) else item for name, item in dict(row).items()}
        hashes.append(hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=False, separators=(',', ':')).encode()).hexdigest())
    return {'count': len(hashes), 'sha256': hashlib.sha256(json.dumps(sorted(hashes), separators=(',', ':')).encode()).hexdigest()}

def audit(label):
    db = sqlite3.connect('file:' + str(DATA / 'marlin.sqlite3') + '?mode=ro', uri=True)
    db.row_factory = sqlite3.Row
    try:
        db.execute('PRAGMA query_only=ON')
        db.execute('BEGIN')
        tables = ('users', 'tenants', 'models', 'model_usage', 'knowledges', 'chunks', 'wiki_pages', 'wiki_folders', 'knowledge_tags', 'task_records', 'desktop_jobs')
        protected = {table: digest(db.execute('SELECT * FROM "' + table + '"')) for table in tables}
        protected['original_knowledge_bases'] = digest(db.execute('SELECT * FROM knowledge_bases WHERE name NOT LIKE ?', (PREFIX + '%',)))
        columns = {row['name'] for row in db.execute('PRAGMA table_info(desktop_jobs)')}
        state = 'status' if 'status' in columns else 'state'
        statuses = {row[0]: row[1] for row in db.execute('SELECT "' + state + '",count(*) FROM desktop_jobs GROUP BY "' + state + '"')}
        files = {}
        for scope in ('.secrets', '.settings', 'media'):
            folder = DATA / scope
            if folder.exists():
                for path in sorted(folder.rglob('*')):
                    if path.is_file():
                        files[str(path.relative_to(DATA))] = hashlib.sha256(path.read_bytes()).hexdigest()
        baseline = json.loads((OLD / 'before-readonly-db.json').read_text())
        unchanged = protected == baseline['protected'] and files == baseline['managed_files_sha256']
        active = sum(count for status, count in statuses.items() if status not in {'completed', 'failed', 'cancelled', 'interrupted'})
        result = {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'label': label, 'data_root': str(DATA), 'read_only': True, 'credential_files_read': False, 'protected': protected, 'managed_files_sha256': files, 'protected_unchanged': unchanged, 'desktop_job_states': statuses, 'active_jobs': active}
        write_new(EV / (label + '-audit.json'), result)
        assert unchanged, 'protected_business_or_managed_files_changed'
        return result
    finally:
        db.close()

def pinned_identity(expected):
    pid = expected['pid']
    proc = Path('/proc') / str(pid)
    fields = (proc / 'stat').read_text().rsplit(')', 1)[1].strip().split()
    assert fields[19] == expected['linux_start_ticks'], 'core_generation_changed'
    assert str((proc / 'exe').resolve()) == expected['exe'], 'core_executable_changed'
    assert str((proc / 'cwd').resolve()) == str(ROOT / 'packages/core'), 'core_working_directory_changed'
    args = (proc / 'cmdline').read_bytes().rstrip(b'\0').decode().split('\0')
    assert args == EXPECTED_ARGS, 'core_exact_arguments_changed'
    return {'pid': pid, 'linux_start_ticks': fields[19], 'exe': expected['exe'], 'cwd': str(ROOT / 'packages/core'), 'exact_args_confirmed': True, 'arguments_or_environment_published': False}

def main():
    p = argparse.ArgumentParser(allow_abbrev=False)
    p.add_argument('phase', choices=['audit', 'stop-owned-core'])
    p.add_argument('label')
    a = p.parse_args()
    assert re.fullmatch(r'[a-z0-9-]{1,50}', a.label)
    result = audit(a.label)
    if a.phase == 'audit':
        print(json.dumps({'label': a.label, 'protected_unchanged': result['protected_unchanged'], 'managed_files': len(result['managed_files_sha256']), 'models': result['protected']['models']['count'], 'usage': result['protected']['model_usage']['count'], 'active_jobs': result['active_jobs']}))
        return
    assert result['active_jobs'] == 0, 'active_task_stop_refused'
    expected = next(row for row in json.loads(CHECKPOINT.read_text())['services'] if row['role'] == 'core')
    identity = pinned_identity(expected)
    descriptor = os.pidfd_open(expected['pid'], 0)
    receipt = {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'identity': identity, 'exact_original_pidfd_retained': True, 'signal': 'SIGTERM', 'stop_attempted': False, 'same_original_pidfd_terminal': False, 'port_or_name_kill': False, 'active_jobs_before': 0}
    try:
        assert pinned_identity(expected) == identity
        write_new(EV / (a.label + '-stop-attempt.json'), receipt)
        signal.pidfd_send_signal(descriptor, signal.SIGTERM, None, 0)
        receipt['stop_attempted'] = True
        start = time.monotonic()
        poll = select.poll()
        poll.register(descriptor, select.POLLIN)
        events = poll.poll(20000)
        receipt['signal_to_original_pidfd_terminal_ms'] = (time.monotonic() - start) * 1000
        receipt['same_original_pidfd_terminal'] = any(fd == descriptor and flags & select.POLLIN for fd, flags in events)
        assert receipt['same_original_pidfd_terminal'], 'owned_core_shutdown_timeout'
    finally:
        os.close(descriptor)
        receipt['original_pidfd_closed'] = True
        write_new(EV / (a.label + '-stop-result.json'), receipt)
        print(json.dumps(receipt))

if __name__ == '__main__':
    main()
