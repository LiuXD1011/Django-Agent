from pathlib import Path
import datetime, hashlib, json, os, subprocess, sys, time

ROOT = Path('/home/liuxuedeng/Django-Agent')
CORE = ROOT / 'packages/core'
NODE = Path('/home/liuxuedeng/.nvm/versions/node/v22.23.2/bin/node')
PARSER = Path('/tmp/marlin-parser-isolation-20261010/bin/python')
COMPAT = Path('/home/liuxuedeng/anaconda3/bin/python')
OUT = ROOT / 'docs/refactor-evidence/20261012-web-search-alias-final-core-checks'
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()

def source_inputs():
    paths = [CORE / name for name in ('package.json', 'package-lock.json', 'tsconfig.json')]
    paths += [path for folder in ('src', 'test', 'scripts') for path in (CORE / folder).rglob('*') if path.is_file() and '__pycache__' not in path.parts]
    assert all(not path.is_symlink() for path in paths)
    return {path.relative_to(ROOT).as_posix(): sha(path) for path in sorted(paths)}

def compiled_outputs():
    paths = [path for path in (CORE / 'dist').rglob('*') if path.is_file()]
    assert paths and all(not path.is_symlink() for path in paths)
    return {path.relative_to(ROOT).as_posix(): sha(path) for path in sorted(paths)}

def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

phase = sys.argv[1]
assert phase in ('freeze', 'full', 'compiled')
assert all(path.is_file() for path in (NODE, PARSER, COMPAT))
pinned = {
    'packages/core/src/settings/settings.ts': 'caad2f1ad67f772482e2dd643d55d855d7ba054ed96874ad15706931149af6ae',
    'packages/core/src/agents/tools.ts': '42dd57fe03c62e937f2735de91838b4d15d92a01edec4d8a75703ca041c252cc',
    'packages/core/test/web-search-settings.test.ts': '12c956a3d5e515b6320cd34ca022082f874cff76f51a67876ab8202846e3a6e7',
    'packages/core/test/jobs.test.ts': 'bdec64f78030c1ade8c0e3c05350f23984cdf037a6c74a91468e3599ef6580fa',
}
assert all(sha(ROOT / name) == digest for name, digest in pinned.items())
freeze = OUT / 'core-source-freeze.json'
if phase == 'freeze':
    OUT.mkdir(exist_ok=False)
    value = {'state': 'frozen', 'createdUtc': utc(), 'sources': source_inputs(), 'compiledOutputs': compiled_outputs(), 'scope': 'final core after endpoint alias guard and real-HTTP deterministic deadline test; no production queue changes in deadline correction', 'previousFreeze': 'docs/refactor-evidence/20261012-web-search-final-core-checks/core-source-freeze.json', 'previousFreezeSHA': sha(ROOT / 'docs/refactor-evidence/20261012-web-search-final-core-checks/core-source-freeze.json')}
    freeze.write_text(json.dumps(value, indent=2))
    print(json.dumps({'state': value['state'], 'sourceCount': len(value['sources']), 'compiledCount': len(value['compiledOutputs']), 'freezeSHA': sha(freeze)}))
    raise SystemExit(0)
frozen = json.loads(freeze.read_text())
assert source_inputs() == frozen['sources'] and compiled_outputs() == frozen['compiledOutputs']
environment = {key: os.environ[key] for key in ('PATH', 'HOME', 'TMPDIR', 'LANG', 'LC_ALL') if key in os.environ}
environment.update(PATH=str(NODE.parent) + os.pathsep + str(COMPAT.parent) + os.pathsep + environment.get('PATH', ''), MARLIN_TEST_PYTHON=str(PARSER), RAGAS_DO_NOT_TRACK='true', HF_HUB_DISABLE_TELEMETRY='1', DO_NOT_TRACK='1')
if phase == 'full':
    compatibility = subprocess.run([str(COMPAT), '-I', '-c', 'import django; assert django.get_version() == "5.2.15"'], env=environment, stdout=subprocess.PIPE, stderr=subprocess.PIPE, check=False)
    assert compatibility.returncode == 0
    files = sorted(path.relative_to(CORE).as_posix() for path in (CORE / 'test').glob('*.test.ts'))
    command = [str(NODE), '--import', 'tsx', '--test', '--test-concurrency=2', *files]
else:
    files = ['test/core-owner-compiled.test.ts']
    command = [str(NODE), '--test', *files]
    environment.update(NODE_OPTIONS='--experimental-strip-types', MARLIN_CORE_OWNER_COMPILED_TEST='1')
log, attempt, receipt = (OUT / (phase + suffix) for suffix in ('.log', '-attempt.json', '-execution.json'))
assert not any(path.exists() for path in (log, attempt, receipt))
value = {'state': 'running', 'phase': phase, 'startedUtc': utc(), 'coreSourceFreezeSHA': sha(freeze), 'command': command, 'cwd': str(CORE), 'fileCount': len(files), 'environmentKeys': sorted(environment), 'pythonCompatibilityExecutable': str(COMPAT), 'auxiliaryParserExecutable': str(PARSER), 'parserDirectoryPrependedToPATH': False, 'compiledOwnerTestsEnabled': phase == 'compiled', 'compiledOnlyTestsRunSeparatelyWithoutTsx': True, 'realUserDataUsed': False, 'realVendorConfigurationUsed': False, 'guiActions': 0}
attempt.write_text(json.dumps(value, indent=2))
print(json.dumps({'state': value['state'], 'phase': phase, 'fileCount': len(files)}), flush=True)
start = time.monotonic()
with log.open('x') as stream:
    result = subprocess.run(command, cwd=CORE, env=environment, stdout=stream, stderr=subprocess.STDOUT, check=False)
value.update(state='completed', exitCode=result.returncode, durationSeconds=time.monotonic()-start, completedUtc=utc(), sourceInputsUnchanged=source_inputs()==frozen['sources'], compiledOutputsUnchanged=compiled_outputs()==frozen['compiledOutputs'], logSHA=sha(log), logBytes=log.stat().st_size)
receipt.write_text(json.dumps(value, indent=2))
print(json.dumps({key: value[key] for key in ('phase', 'exitCode', 'durationSeconds', 'sourceInputsUnchanged', 'compiledOutputsUnchanged', 'logSHA')}), flush=True)
raise SystemExit(result.returncode)
