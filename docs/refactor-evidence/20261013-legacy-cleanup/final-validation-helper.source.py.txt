import ast
import hashlib
import json
import re
import subprocess
from datetime import datetime, timezone
from pathlib import Path
from urllib.parse import unquote, urlparse


ROOT = Path('/home/liuxuedeng/Django-Agent')
EVIDENCE = ROOT / 'docs/refactor-evidence/20261013-legacy-cleanup'
OUTPUT = EVIDENCE / 'final-validation.json'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git(*args):
    result = subprocess.run(['git', *args], cwd=ROOT, text=True, capture_output=True)
    assert result.returncode == 0, result.stderr
    return result.stdout


def check_command(command):
    result = subprocess.run(command, cwd=ROOT, text=True, capture_output=True)
    assert result.returncode == 0, result.stdout + result.stderr
    return {'command': command, 'actual_exit_code': result.returncode,
            'stdout': result.stdout, 'stderr': result.stderr}


deletion = json.loads((EVIDENCE / 'deletion.json').read_text())
files = deletion['files']
assert len(files) == deletion['candidate_count'] == 301
assert deletion['actual_exit_code'] == 0
assert all(not (ROOT / item['path']).exists() for item in files)
assert all(sha(ROOT / item['archive']) == item['sha256'] for item in files)
bytecode_groups = [items for items in deletion.values()
                   if isinstance(items, list) and items and isinstance(items[0], dict)
                   and items[0].get('path', '').endswith('.pyc')]
assert len(bytecode_groups) == 1 and len(bytecode_groups[0]) == 248
assert all(not (ROOT / item['path']).exists() for item in bytecode_groups[0])
orphan = json.loads((EVIDENCE / 'orphan-legacy-cache-cleanup.json').read_text())
assert orphan['deleted'] and not (ROOT / orphan['path']).exists()
old_roots = ['accounts', 'agent', 'chat', 'config', 'core', 'knowledge',
             'models_config', 'personal_knowledge_base', 'wiki']
assert all(not (ROOT / name).exists() for name in old_roots)
assert set(git('diff', '--diff-filter=D', '--name-only').splitlines()) == {
    item['path'] for item in files}
changed = git('diff', '--name-only').splitlines()
assert all(path in {item['path'] for item in files}
           or path in {'.gitignore', 'README.md', 'tests/test_frontend_contracts.py'}
           or path.startswith('docs/') for path in changed)
retained = ['LICENSE', 'workers/document/requirements.txt', 'workers/document/LICENSE',
            'workers/document/worker.py', 'workers/document/tests/fixtures/manifest.json',
            'workers/document/tests/fixtures/legacy/LICENSE',
            'workers/evaluation/requirements.txt', 'workers/evaluation/LICENSE',
            'workers/evaluation/worker.py', 'packages/core/src/auth/passwords.ts',
            'packages/core/src/storage/legacy.ts', 'packages/core/src/legacy/index.ts',
            'desktop/native-windows.lock.json']
assert all((ROOT / path).is_file() for path in retained)
ignore_patterns = lambda text: [line for line in text.splitlines()
                                if line.strip() and not line.lstrip().startswith('#')]
assert ignore_patterns(git('show', 'HEAD:.gitignore')) == ignore_patterns(
    (ROOT / '.gitignore').read_text())
test_path = 'tests/test_frontend_contracts.py'
test_class = lambda text: next(node for node in ast.parse(text).body
                              if isinstance(node, ast.ClassDef)
                              and node.name == 'FrontendContractTests')
assert ast.dump(test_class(git('show', 'HEAD:' + test_path))) == ast.dump(
    test_class((ROOT / test_path).read_text()))
source = json.loads((EVIDENCE / 'core-source-reuse.json').read_text())
assert source['current_source_count'] == source['formal_source_count'] == 483
assert source['source_before_after_equal'] and source['formal_source_equal']
assert source['added'] == source['removed'] == source['changed'] == []
assert sha(ROOT / source['formal_report']) == source['formal_report_sha256']
formal_inputs = json.loads((ROOT / source['formal_report']).read_text())['source_sha256']
assert len(formal_inputs) == 483
assert all((ROOT / path).is_file() and sha(ROOT / path) == expected
           for path, expected in formal_inputs.items())
for receipt, key in [('root-helper-cleanup.json', 'helpers'),
                     ('core-consumed-helpers.json', 'files')]:
    data = json.loads((EVIDENCE / receipt).read_text())
    assert all(not Path(item['path']).exists() for item in data[key])
report = ROOT / 'docs/desktop-refactor/LEGACY-CLEANUP.md'
local_links = []
for match in re.finditer(r'(?<!!)\[[^\]]+\]\(([^)]+)\)', report.read_text()):
    target = match.group(1).strip()
    parsed = urlparse(target)
    if parsed.scheme or target.startswith('#'):
        continue
    path = (report.parent / unquote(parsed.path)).resolve()
    assert path == OUTPUT or path.exists(), target
    local_links.append(target)
checks = [check_command(['python', '-B', 'tests/test_readme_links.py']),
          check_command(['git', 'diff', '--check'])]
self_path = Path(__file__)
archive = EVIDENCE / 'final-validation-helper.source.py.txt'
archive.write_bytes(self_path.read_bytes())
receipt = {
    'utc': datetime.now(timezone.utc).isoformat(), 'actual_exit_code': 0,
    'deleted_candidate_count': 301, 'deleted_old_bytecode_count': 249,
    'absent_old_package_roots': old_roots, 'archive_sha_matches': 301,
    'tracked_deletions_match_exact_receipt': True,
    'tracked_changes_confined_to_authorized_scope': True,
    'retained_required_files': retained, 'ignore_patterns_unchanged': True,
    'frontend_contract_assertions_unchanged': True,
    'source_reuse_receipt_verified': True, 'formal_source_count': 483,
    'current_formal_input_hashes_verified': 483,
    'report_local_links_checked': local_links, 'checks': checks,
    'prior_owned_helpers_absent': True,
    'final_helper': {'path': str(self_path), 'archive': archive.name,
                     'sha256': sha(archive), 'deleted': True},
    'user_data_or_secrets_read': False, 'user_data_or_secrets_deleted': False,
    'commit_push_or_release_performed': False,
    'limits': ['Targeted checks only; no new Windows build or GUI/model acceptance.']
}
self_path.unlink()
assert not self_path.exists()
OUTPUT.write_text(json.dumps(receipt, ensure_ascii=False, indent=2) + '\n')
assert all((report.parent / unquote(urlparse(target).path)).resolve().exists()
           for target in local_links)
print(json.dumps({'deleted_files': 301, 'deleted_bytecode': 249,
                  'old_package_roots_absent': len(old_roots),
                  'archive_hash_matches': 301, 'checks': checks,
                  'helper_removed': True, 'receipt': str(OUTPUT)},
                 ensure_ascii=False))
