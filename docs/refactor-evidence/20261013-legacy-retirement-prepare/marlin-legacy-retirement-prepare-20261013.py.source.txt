def run(label, args):
    p = subprocess.run(args, capture_output=True, text=True)
    (OUT / (label + '.stdout.log')).write_text(p.stdout)
    (OUT / (label + '.stderr.log')).write_text(p.stderr)
    commands.append({'label':label,'argv':args,'exit_code':p.returncode})
    if p.returncode != 0:
        raise RuntimeError(label)
    return p.stdout

def domain(path):
    if path.startswith(('config/','scripts/','manage.py','requirements.txt','desktop/requirements','core/')): return 'startup'
    if '/migrations/' in path or re.search('credential|models.py|db_utils|db_retry',path): return 'storage'
    if path.startswith('accounts/') or re.search('authentication|permission|personal_space|view_security',path): return 'auth'
    if path.startswith('models_config/') or re.search('model_|thinking|llm_provider|integration_config|capabilities',path): return 'settings'
    if path.startswith('wiki/') or re.search('wiki_',path): return 'wiki'
    if path.startswith('agent/'): return 'legacy-channels' if Path(path).name in {'urls.py','views.py'} else 'agents'
    if re.search('eval|ragas|open_rag|benchmark',path): return 'evaluation'
    if re.search('document|chunking|chunk_|multimodal|testdata',path): return 'documents'
    if re.search('search|retrieval|rag_pipeline|parent_context|query_understand|hybrid',path): return 'retrieval'
    if re.search('graph|neo4j',path): return 'graph'
    if re.search('memory|consolidator',path): return 'memory'
    if re.search('task|stream|span_|event_log|runtime|sqlite_concurrency|business_publish',path): return 'jobs'
    if re.search('langfuse|observability',path): return 'integrations'
    if path.startswith('knowledge/') or re.search('knowledge_cleanup|legacy_cleanup',path): return 'knowledge'
    if re.search('agent_|actor|prompt|context_|reasoning|tool_',path): return 'agents'
    return 'chat'
