from pathlib import Path
from datetime import datetime, timezone
import difflib
import hashlib
import json
import re
import shutil

ROOT = Path('/home/liuxuedeng/Django-Agent')
ORIGINAL = ROOT / 'docs/refactor-evidence/20261013-asar-bootstrap-runtime-review'
OUT = ORIGINAL / 'iteration-2'
SELF = Path(__file__).resolve()

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def write(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

shutil.copyfile(SELF, OUT / 'finalize-helper.source.py.txt')
old_path = ROOT / 'docs/refactor-evidence/20261013-asar-bootstrap-runtime-cleanup/before/packaged-bootstrap.test.cjs'
new_path = OUT / 'reviewed-test.source.cjs.txt'
assert sha(old_path) == '184ca9c855368a7543b1f4cd0ba31dd72a03475410b9592332c59f8abb3dc813'
assert sha(new_path) == '46cdd96a9b4b625d0478a209b627204dc7bb8845981645695130c945cf939272'
assert sha(ROOT / 'desktop/packaged-bootstrap.test.cjs') == sha(new_path)
assert sha(ROOT / 'desktop/runtime.cjs') == 'fb2b569bbb94c4237608b515d97f78914cc996322116405c1d009244414b9112'
diff = ''.join(difflib.unified_diff(old_path.read_text().splitlines(keepends=True), new_path.read_text().splitlines(keepends=True),
    fromfile='before/desktop/packaged-bootstrap.test.cjs', tofile='after/desktop/packaged-bootstrap.test.cjs'))
(OUT / 'cleanup-test.diff').write_text(diff, encoding='utf-8')
original = json.loads((OUT / 'original-evidence-preserved-before.json').read_text())
current = {str(p.relative_to(ORIGINAL)): sha(p) for p in ORIGINAL.rglob('*') if p.is_file() and OUT not in p.parents}
assert original == current
links = []
for target in re.findall(r'\]\(([^\n)]+)\)', (OUT / 'REPORT.md').read_text()):
    if target.startswith(('https:', 'http:', '#')):
        continue
    path = (OUT / target.split('#', 1)[0]).resolve()
    assert path.is_file(), f'Missing link: {target}'
    links.append({'target': target, 'sha256': sha(path)})
write('final-validation.json', {'actualExitCode': 0, 'relativeLinks': len(links), 'links': links,
    'oldReviewEvidenceFilesUnchanged': len(original), 'newTestSHA': sha(new_path), 'R01State': 'closed',
    'sourceRuntimeUnchanged': True, 'productionBlockingFindings': 0, 'newFindings': 0,
    'oldInstalledNormalGUIStartupStillFail': True, 'utc': datetime.now(timezone.utc).isoformat()})
own_hash = sha(SELF)
assert own_hash == sha(OUT / 'finalize-helper.source.py.txt')
SELF.unlink()
write('finalize-helper-cleanup.json', {'exactOwnedPath': str(SELF), 'sha256': own_hash, 'sourceArchiveEqual': True, 'removed': not SELF.exists()})
files = [{'path': str(p.relative_to(OUT)), 'bytes': p.stat().st_size, 'sha256': sha(p)} for p in sorted(OUT.rglob('*')) if p.is_file() and p.name != 'freeze.json']
write('freeze.json', {'scope': 'R-01 iteration-2 independent review only; original review evidence unchanged', 'files': files})
print(json.dumps({'actualExitCode': 0, 'reportSHA': sha(OUT / 'REPORT.md'), 'freezeSHA': sha(OUT / 'freeze.json'),
    'relativeLinks': len(links), 'oldEvidenceFilesUnchanged': len(original), 'R01State': 'closed',
    'temporaryHelperRemoved': not SELF.exists()}))
