def checked(value, message):
    if not value:
        raise RuntimeError(f'{message}: Win32 {ctypes.get_last_error()}')
    return value

identity_cache = {}

def projection(handle):
    times = [wintypes.FILETIME() for _ in range(4)]
    checked(kernel.GetProcessTimes(handle, *(ctypes.byref(value) for value in times)), 'Read original process FILETIME')
    capacity = wintypes.DWORD(32768)
    image = ctypes.create_unicode_buffer(capacity.value)
    if handle not in identity_cache:
        checked(kernel.QueryFullProcessImageNameW(handle, 0, image, ctypes.byref(capacity)), 'Read live original process image')
        identity_cache[handle] = image.value
    else:
        image.value = identity_cache[handle]
    wait = kernel.WaitForSingleObject(handle, 0)
    assert wait in {0, 258}, 'Original process observation failed'
    code = wintypes.DWORD()
    if wait == 0:
        checked(kernel.GetExitCodeProcess(handle, ctypes.byref(code)), 'Read real terminal code')
    return {'pid': kernel.GetProcessId(handle), 'createdAt': str((times[0].dwHighDateTime << 32) | times[0].dwLowDateTime),
            'executable': image.value, 'wait': wait, 'exited': wait == 0,
            'exitCode': code.value if wait == 0 else None}

handles = {}
observer_records = []
observer_errors = []
stop = threading.Event()

def observer():
    consumed = set()
    while not stop.is_set():
        for request_path in sorted(EXCHANGE.glob('*.request.json')):
            if request_path.name in consumed:
                continue
            consumed.add(request_path.name)
            request = json.loads(request_path.read_text(encoding='utf-8'))
            try:
                result = {'ok': True, 'operation': request['operation'], 'label': request['label'], 'members': []}
                if request['operation'] == 'observe':
                    for member in request['members']:
                        pid = member['pid']
                        assert isinstance(pid, int) and pid > 0
                        handle = checked(kernel.OpenProcess(0x00101000, False, pid), 'Open only a reported owned fixture for observation')
                        try:
                            actual = projection(handle)
                            assert actual['pid'] == pid and not actual['exited']
                            assert actual['executable'].casefold() == member['executable'].casefold()
                            assert member['createdAt'] is None or actual['createdAt'] == member['createdAt']
                            key = (request['label'], pid)
                            assert key not in handles
                            handles[key] = handle
                            handle = None
                            result['members'].append(actual)
                        finally:
                            if handle:
                                kernel.CloseHandle(handle)
                elif request['operation'] == 'check':
                    for expected_exited, pids in ((True, request['exited']), (False, request['alive'])):
                        for pid in pids:
                            handle = handles[(request['label'], pid)]
                            if expected_exited:
                                assert kernel.WaitForSingleObject(handle, 5000) == 0, 'Original fixture HANDLE did not terminate'
                            actual = projection(handle)
                            assert actual['exited'] == expected_exited, 'Original fixture HANDLE state differs from requirement'
                            result['members'].append(actual)
                else:
                    raise RuntimeError('Unexpected observer request')
            except Exception as error:
                result = {'ok': False, 'error': str(error), 'label': request.get('label')}
                observer_errors.append(result)
            observer_records.append(result)
            response = request_path.with_name(request_path.name.replace('.request.json', '.response.json'))
            temporary = response.with_suffix('.tmp')
            temporary.write_text(json.dumps(result, ensure_ascii=False, indent=2), encoding='utf-8')
            temporary.replace(response)
        stop.wait(0.01)
