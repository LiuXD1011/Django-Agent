class Accounting(ctypes.Structure):
    _fields_=[(name,ctypes.c_longlong) for name in ('user','kernel','period_user','period_kernel')]+[(name,ctypes.c_uint32) for name in ('faults','total','active','terminated')]

class Kernel:
    def __init__(self,signal_ms=0,child_ms=0,exit_code=0,wait_failure=None,accounting_failure=False,exit_failure=False):
        self.ms=0;self.signal_ms=signal_ms;self.child_ms=child_ms;self.exit=exit_code;self.wait_failure=wait_failure;self.accounting_failure=accounting_failure;self.exit_failure=exit_failure;self.last_error=0;self.waits=[];self.query_count=0;self.exit_reads=0;self.terminations=0
    def WaitForSingleObject(self,handle,timeout):
        assert handle==0x1234;self.waits.append(timeout)
        if self.wait_failure is not None:self.last_error=self.wait_failure;return 0xffffffff
        if self.ms>=self.signal_ms:return 0
        self.ms+=min(timeout,self.signal_ms-self.ms)
        return 0 if self.ms>=self.signal_ms else 258
    def QueryInformationJobObject(self,job,kind,pointer,size,returned):
        assert job==0x5678 and kind==1 and size==48;self.query_count+=1
        if self.accounting_failure:self.last_error=6;return False
        pointer._obj.active=1 if self.ms<self.child_ms else 0;return True
    def GetExitCodeProcess(self,handle,pointer):
        assert handle==0x1234;self.exit_reads+=1
        if self.exit_failure:self.last_error=5;return False
        pointer._obj.value=self.exit;return True
    def TerminateProcess(self,*args):self.terminations+=1;raise AssertionError('termination prohibited')
    def TerminateJobObject(self,*args):self.terminations+=1;raise AssertionError('termination prohibited')
    def sleep(self,seconds):self.ms+=seconds*1000

class ReapTests(unittest.TestCase):
    def invoke(self,k):
        old_time=getattr(h,'time',None);old_error=getattr(ctypes,'get_last_error',None)
        h.time=SimpleNamespace(monotonic=lambda:k.ms/1000,sleep=k.sleep);ctypes.get_last_error=lambda:k.last_error
        capture=io.StringIO()
        try:
            with contextlib.redirect_stdout(capture):
                fn=legacy_boundary() if OLD else h.reap_owned_bootstrap
                return fn((k,SimpleNamespace(DWORD=ctypes.c_uint32),None,Accounting,None,None),0x1234,0x5678)
        finally:
            h.time=old_time
            if old_error is None:del ctypes.get_last_error
            else:ctypes.get_last_error=old_error
            self.assertEqual(k.terminations,0)
            self.observed_stdout=capture.getvalue()
    def test_immediate_exit_zero(self):
        self.assertEqual(self.invoke(Kernel())['bootstrap_exit_code'],0)
    def test_six_second_kernel_signal_is_reaped_within_fixed_budget(self):
        k=Kernel(signal_ms=6000)
        try:result=self.invoke(k)
        except (h.Refusal,legacy.Refusal) as error:self.fail('actual five-second guard rejects eligible delayed signal: '+str(error))
        self.assertEqual(result['bootstrap_exit_code'],0);self.assertGreaterEqual(k.ms,6000);self.assertLessEqual(k.ms,30000)
    def test_signalled_bootstrap_does_not_accept_active_owned_children(self):

    def test_nonzero_bootstrap_exit_refused(self):
