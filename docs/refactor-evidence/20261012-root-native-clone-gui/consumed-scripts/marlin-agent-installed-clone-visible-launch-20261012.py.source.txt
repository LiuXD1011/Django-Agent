def digest(path):

def checked_identity_api(native,handle,pid,exe,operation,succeeded):
    if succeeded:return True
    # Save the failing API's thread-local error BEFORE any other Win32 call.
    error=ctypes.get_last_error();k,w=native[:2]
    wait=int(k.WaitForSingleObject(handle,0));wait_error=ctypes.get_last_error() if wait==0xffffffff else 0
    code=w.DWORD();available=bool(k.GetExitCodeProcess(handle,ctypes.byref(code)))
    exit_error=ctypes.get_last_error() if not available else 0
    exited=wait==0 and available
    RACES.append({'candidate_pid':int(pid),'candidate_exe_basename':str(exe),'api':operation,'api_last_error':error,'same_handle_wait_after':wait,'wait_last_error':wait_error,'exit_code_available':available,'exit_code':int(code.value) if available else None,'exit_code_last_error':exit_error,'ignored_proven_exit':exited})
