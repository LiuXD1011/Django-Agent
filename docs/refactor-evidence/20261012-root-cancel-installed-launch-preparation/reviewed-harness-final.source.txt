class Refusal(Exception):
    """Fixed safe codes only. Private values must never enter public receipts."""


def need(value, code):
    if not value:
        raise Refusal(code)


def digest(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def json_hash(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def safe_path(path):
    for candidate in (Path(path), *Path(path).parents):
        try:
            info = candidate.lstat()
        except FileNotFoundError:
            continue
        need(not stat.S_ISLNK(info.st_mode) and not getattr(info, "st_file_attributes", 0) & 0x400, "reparse_path_rejected")


def inventory(root):
    safe_path(root)
    need(Path(root).is_dir(), "inventory_directory_missing")
    result = {}

    def visit(folder):
        with os.scandir(folder) as entries:
            for entry in sorted(entries, key=lambda item: item.name):
                info = entry.stat(follow_symlinks=False)
                need(not stat.S_ISLNK(info.st_mode) and not getattr(info, "st_file_attributes", 0) & 0x400, "reparse_inventory_rejected")
                file = Path(entry.path)
                if stat.S_ISDIR(info.st_mode):
                    visit(file)
                else:
                    need(stat.S_ISREG(info.st_mode), "nonregular_inventory_file")
                    result[file.relative_to(root).as_posix()] = digest(file)

    visit(Path(root))
    return result


def load(path):
    safe_path(path)
    need(Path(path).is_file(), "required_json_missing")
    value = json.loads(Path(path).read_text(encoding="utf-8-sig"))
    need(isinstance(value, dict), "required_json_invalid")
    return value


def write_new(path, value):
    safe_path(path)
    with Path(path).open("x", encoding="utf-8", newline="\n") as stream:
        stream.write(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def marker(path, value):

def native_api():
    from ctypes import wintypes as w
    k = ctypes.WinDLL("kernel32", use_last_error=True)
    size = ctypes.c_size_t

    class Basic(ctypes.Structure):
        _fields_ = [("user", ctypes.c_longlong), ("job", ctypes.c_longlong), ("flags", w.DWORD), ("min", size), ("max", size), ("limit", w.DWORD), ("affinity", size), ("priority", w.DWORD), ("scheduling", w.DWORD)]

    class Io(ctypes.Structure):
        _fields_ = [(name, ctypes.c_ulonglong) for name in ("reads", "writes", "other", "read_bytes", "write_bytes", "other_bytes")]

    class Extended(ctypes.Structure):
        _fields_ = [("basic", Basic), ("io", Io)] + [(name, size) for name in ("process", "job", "peak_process", "peak_job")]

    class Accounting(ctypes.Structure):
        _fields_ = [(name, ctypes.c_longlong) for name in ("user", "kernel", "period_user", "period_kernel")] + [(name, w.DWORD) for name in ("faults", "total", "active", "terminated")]

    class Startup(ctypes.Structure):
        _fields_ = [("cb", w.DWORD), ("reserved", w.LPWSTR), ("desktop", w.LPWSTR), ("title", w.LPWSTR)] + [(name, w.DWORD) for name in ("x", "y", "width", "height", "columns", "rows", "fill", "flags")] + [("show", w.WORD), ("reserved_size", w.WORD), ("reserved_bytes", ctypes.c_void_p), ("stdin", w.HANDLE), ("stdout", w.HANDLE), ("stderr", w.HANDLE)]

    class Process(ctypes.Structure):
        _fields_ = [("process", w.HANDLE), ("thread", w.HANDLE), ("pid", w.DWORD), ("tid", w.DWORD)]

    signatures = {
        "CreateJobObjectW": ([ctypes.c_void_p, w.LPCWSTR], w.HANDLE),
        "SetInformationJobObject": ([w.HANDLE, ctypes.c_int, ctypes.c_void_p, w.DWORD], w.BOOL),
        "QueryInformationJobObject": ([w.HANDLE, ctypes.c_int, ctypes.c_void_p, w.DWORD, ctypes.c_void_p], w.BOOL),
        "AssignProcessToJobObject": ([w.HANDLE, w.HANDLE], w.BOOL),
        "TerminateJobObject": ([w.HANDLE, w.UINT], w.BOOL),
        "TerminateProcess": ([w.HANDLE, w.UINT], w.BOOL),
        "CreateProcessW": ([w.LPCWSTR, w.LPWSTR, ctypes.c_void_p, ctypes.c_void_p, w.BOOL, w.DWORD, ctypes.c_void_p, w.LPCWSTR, ctypes.POINTER(Startup), ctypes.POINTER(Process)], w.BOOL),
        "ResumeThread": ([w.HANDLE], w.DWORD),
        "WaitForSingleObject": ([w.HANDLE, w.DWORD], w.DWORD),
        "GetExitCodeProcess": ([w.HANDLE, ctypes.POINTER(w.DWORD)], w.BOOL),
        "CloseHandle": ([w.HANDLE], w.BOOL),
        "OpenProcess": ([w.DWORD, w.BOOL, w.DWORD], w.HANDLE),
        "QueryFullProcessImageNameW": ([w.HANDLE, w.DWORD, w.LPWSTR, ctypes.POINTER(w.DWORD)], w.BOOL),
        "GetProcessTimes": ([w.HANDLE, ctypes.POINTER(w.FILETIME), ctypes.POINTER(w.FILETIME), ctypes.POINTER(w.FILETIME), ctypes.POINTER(w.FILETIME)], w.BOOL),
        "CreateFileW": ([w.LPCWSTR, w.DWORD, w.DWORD, ctypes.c_void_p, w.DWORD, w.DWORD, w.HANDLE], w.HANDLE),
    }
    for name, (arguments, result) in signatures.items():
        getattr(k, name).argtypes, getattr(k, name).restype = arguments, result
    return k, w, Extended, Accounting, Startup, Process


def process_handle(native, pid):
    k, w, *_ = native
    handle = k.OpenProcess(0x00101000, False, pid)
    need(handle, "protected_process_unavailable")
    try:
        image = ctypes.create_unicode_buffer(32768)
        length = w.DWORD(len(image))
        need(k.QueryFullProcessImageNameW(handle, 0, image, ctypes.byref(length)), "protected_process_path_unavailable")
        creation, exit_time, kernel, user = (w.FILETIME() for _ in range(4))
        need(k.GetProcessTimes(handle, ctypes.byref(creation), ctypes.byref(exit_time), ctypes.byref(kernel), ctypes.byref(user)), "protected_process_creation_unavailable")
        need(k.WaitForSingleObject(handle, 0) == 258, "protected_process_already_exited")
        identity = {"pid": pid, "image": image.value, "creation_filetime": (creation.dwHighDateTime << 32) | creation.dwLowDateTime}
        return handle, identity
    except BaseException:
        k.CloseHandle(handle)
        raise



def registry_state():
    import winreg
    rows = []
    suffixes = ("Software\\" + GUID, "Software\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\" + GUID)
    for view, view_name in ((winreg.KEY_WOW64_64KEY, "64"), (winreg.KEY_WOW64_32KEY, "32")):
        for hive, name in ((winreg.HKEY_LOCAL_MACHINE, "HKLM"), (winreg.HKEY_CURRENT_USER, "HKCU")):
            for index, suffix in enumerate(suffixes):
                row = {"hive": name, "view": view_name, "key": suffix, "exists": False}
                try:
                    with winreg.OpenKey(hive, suffix, 0, winreg.KEY_READ | view) as key:
                        need(name == "HKCU", "machine_installation_refused")
                        if index == 0:
                            value = winreg.QueryValueEx(key, "InstallLocation")[0]
                            need(isinstance(value, str) and value.casefold() == str(INSTALL).casefold(), "foreign_install_location")
                            row["install_location"] = value
                        else:
                            need(winreg.QueryValueEx(key, "UninstallString")[0] == '"' + str(INSTALL / "Uninstall Marlin.exe") + '" /currentuser', "foreign_uninstaller_registration")
                            need(winreg.QueryValueEx(key, "QuietUninstallString")[0] == '"' + str(INSTALL / "Uninstall Marlin.exe") + '" /currentuser /S', "foreign_quiet_uninstaller_registration")
                            row["uninstall_executable"] = str(INSTALL / "Uninstall Marlin.exe")
                        row["exists"] = True
                except FileNotFoundError:
                    need(name == "HKLM", "owned_registration_missing")
                rows.append(row)
    return rows

def read_asar_assets(archive, audit):
    """Read the verified fixed flat ASAR. Layout read from installed @electron/asar.

    Eight-byte size Pickle, header Pickle containing one string, followed by
    bytes at 8+headerSize+offset. No general extraction or arbitrary filename.
    """
    safe_path(archive)
    with archive.open("rb") as stream:
        prefix = stream.read(8)
        need(len(prefix) == 8, "asar_prefix_short")
        payload_size, header_size = struct.unpack("<II", prefix)
        need(payload_size == 4 and 8 <= header_size <= 1024 * 1024, "asar_size_pickle_invalid")
        header = stream.read(header_size)
        need(len(header) == header_size, "asar_header_short")
        header_payload, string_size = struct.unpack_from("<II", header)
        need(header_payload == header_size - 4 and 0 < string_size <= header_payload - 4, "asar_header_pickle_invalid")
        parsed = json.loads(header[8:8 + string_size].decode("utf-8"))
        entries = parsed.get("files")
        need(isinstance(entries, dict) and sorted(entries) == sorted(audit["asar_entries"]), "asar_fixed_allowlist_changed")
        result = {}
        for name, info in entries.items():
            need(isinstance(info, dict) and not any(key in info for key in ("link", "unpacked", "files")), "asar_nonpacked_or_nested_asset")
            size = info.get("size"); offset = info.get("offset")
            need(isinstance(size, int) and not isinstance(size, bool) and 0 <= size <= 1024 * 1024 and isinstance(offset, str) and offset.isdecimal(), "asar_asset_range_invalid")
            start = 8 + header_size + int(offset)
            need(start + size <= archive.stat().st_size, "asar_asset_outside_archive")
            stream.seek(start); content = stream.read(size)
            need(len(content) == size and hashlib.sha256(content).hexdigest() == audit["asset_sha256"][name], "asar_asset_hash_changed")
            result[name] = content
        return result

def retained_state(require_startup_ready=True):
    # Caller must own the deny-write/delete handle and have checked zero WAL.
    # Never decrypt, report row values, write schema, refresh or initialize core.
    conn = sqlite3.connect((DATA / "marlin.sqlite3").as_uri() + "?mode=ro&immutable=1", uri=True)
    try:
        conn.execute("PRAGMA query_only=ON")
        need(conn.execute("PRAGMA integrity_check").fetchone()[0] == "ok", "retained_integrity_failed")
        need(not conn.execute("PRAGMA foreign_key_check").fetchall(), "retained_foreign_keys_failed")
        rows = {}
        for name in TABLES:
            values = list(conn.execute('SELECT * FROM "' + name + '" ORDER BY id'))
            rows[name] = {"count": len(values), "rows_sha256": json_hash(values)}
        user = conn.execute("SELECT tenant_id FROM users WHERE id=? AND is_active=1 AND deleted_at IS NULL", (USER,)).fetchone()
        kb = conn.execute("SELECT tenant_id FROM knowledge_bases WHERE id=? AND deleted_at IS NULL", (KB,)).fetchone()
        need(user and kb and str(user[0]) == str(kb[0]), "retained_user_or_library_missing")
        saved = conn.execute("SELECT value FROM desktop_meta WHERE key='ipc.session'").fetchone()
        if require_startup_ready:
            need(saved, "private_saved_session_missing")
        if saved:
            envelope = json.loads(saved[0])
            need(isinstance(envelope, dict) and isinstance(envelope.get("token"), str) and envelope["token"].startswith("enc:v1:"), "private_session_envelope_unavailable")
        need(conn.execute("SELECT count(*) FROM desktop_meta WHERE key='ipc.receipts'").fetchone()[0] == 0, "unsettled_auth_receipts_refused")
        if require_startup_ready:
            need(conn.execute("SELECT count(*) FROM desktop_jobs WHERE status IN ('queued','running')").fetchone()[0] == 0 and conn.execute("SELECT count(*) FROM task_records WHERE status IN ('queued','running')").fetchone()[0] == 0, "automatic_job_recovery_refused")
    finally:
        conn.close()
    key = DATA / ".secrets/credentials.key"
    safe_path(key)
    need(key.is_file() and key.stat().st_size == 32, "retained_private_key_unavailable")
    credentials = inventory(DATA / ".secrets")
    return {"business_tables": rows, "credential_material_sha256": credentials, "data_files_sha256": inventory(DATA)}

def run_owned_tree(native, name, identity, on_started=None):
    """Suspend -> own private Job -> resume. Reap all children, never image-kill.

    Visible execution keeps this harness/Job alive while root uses the GUI.
    Only its own suspended bootstrap or private Job may be stopped on failure.
    """
    import msvcrt
    executable, arguments, timeout = fixed_execution(name)
    safe_path(executable)
    need(executable.is_file(), "fixed_executable_missing")
    executable_key = "Marlin.exe" if name in ("hidden-smoke", "visible-retained") else "node/node.exe" if name == "native-ipc" else "python/python.exe"
    need(digest(executable) == identity["executables_sha256"][executable_key], "fixed_executable_hash_changed_at_launch")
    k, w, Extended, Accounting, Startup, Process = native
    process = Process(); job = k.CreateJobObjectW(None, None)
    need(job, "private_job_create_failed")
    assigned = completed = False
    started = time.monotonic()
    command = subprocess.list2cmdline([str(executable), *arguments])
    log = WORK / (name + ".log")
    safe_path(log)
    try:
        limits = Extended(); limits.basic.flags = 0x2000
        need(k.SetInformationJobObject(job, 9, ctypes.byref(limits), ctypes.sizeof(limits)), "private_job_limits_failed")
        # Retained-data stdout/stderr go to NUL: no private auth errors in logs.
        with (open(os.devnull, "wb") if name == "visible-retained" else log.open("xb")) as output, open(os.devnull, "rb") as input_stream:
            handles = [msvcrt.get_osfhandle(stream.fileno()) for stream in (input_stream, output)]
            for handle in handles:
                os.set_handle_inheritable(handle, True)
            try:
                startup = Startup(); startup.cb = ctypes.sizeof(startup); startup.flags = 0x100
                startup.stdin, startup.stdout, startup.stderr = handles[0], handles[1], handles[1]
                environment = ctypes.create_unicode_buffer("\0".join(key + "=" + value for key, value in sorted(minimal_env().items(), key=lambda item: item[0].casefold())) + "\0\0")
                need(k.CreateProcessW(str(executable), ctypes.create_unicode_buffer(command), None, None, True, 0x4 | 0x400 | 0x08000000, environment, str(OWNED), ctypes.byref(startup), ctypes.byref(process)), "owned_process_create_failed")
            finally:
                for handle in handles:
                    os.set_handle_inheritable(handle, False)
            need(k.AssignProcessToJobObject(job, process.process), "owned_process_assign_failed")
