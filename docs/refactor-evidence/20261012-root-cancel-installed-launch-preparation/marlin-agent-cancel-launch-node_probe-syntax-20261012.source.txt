"use strict";
const assert = require("node:assert/strict"), fs = require("node:fs"), path = require("node:path");
const {createRequire} = require("node:module"), {pathToFileURL} = require("node:url");
const [root, runtimeCopy, data, output] = process.argv.slice(2);
let runtime, step = "native";
async function main() {
  assert.equal(process.platform,"win32"); assert.equal(process.arch,"x64"); assert.equal(process.version,"v22.23.2");
  assert.equal(path.resolve(process.execPath),path.join(root,"node","node.exe"));
  assert.equal(process.env.PATH,"C:\\Windows\\System32");
  const core = path.join(root,"core"), requireCore = createRequire(path.join(core,"package.json"));
  for (const name of ["better-sqlite3","sqlite-vec"]) assert.ok(requireCore.resolve(name).startsWith(path.join(core,"node_modules")+path.sep));
  const {openNativeDatabase} = await import(pathToFileURL(path.join(core,"dist/storage/database.js")).href);
  const native = openNativeDatabase(":memory:"); let nativeResult;
  try {
    native.defaultSafeIntegers(); requireCore("sqlite-vec").load(native);
    const id=(1n<<60n)+321n;
    native.exec("CREATE VIRTUAL TABLE f USING fts5(content); CREATE VIRTUAL TABLE v USING vec0(embedding float[2])");
    native.prepare("INSERT INTO f(rowid,content) VALUES (?,?)").run(id,"installed native packaging verification");
    native.prepare("INSERT INTO v(rowid,embedding) VALUES (?,?)").run(id,new Float32Array([0.25,0.75]));
    const f=native.prepare("SELECT rowid FROM f WHERE f MATCH 'packaging'").get();
    const v=native.prepare("SELECT rowid,distance FROM v WHERE embedding MATCH ? AND k=1").get(new Float32Array([0.25,0.75]));
    assert.equal(f.rowid,id); assert.equal(v.rowid,id); assert.equal(v.distance,0);
    nativeResult={sqlite_version:native.prepare("SELECT sqlite_version() version").get().version,vector_version:native.prepare("SELECT vec_version() version").get().version,fts_id:f.rowid.toString(),vector_id:v.rowid.toString()};
  } finally { native.close(); }
  step="private_fork";
  assert.deepEqual(fs.readdirSync(data),[]);
  const {Runtime}=require(runtimeCopy); let helperFailed=false;
  runtime=new Runtime(root,data,()=>{helperFailed=true;});
  const started=performance.now(), ready=await runtime.start();
  assert.equal(ready.protocol,1); assert.equal(ready.runtime,"typescript"); assert.equal(ready.http_listener,false); assert.equal(ready.version,"0.7.0");
  assert.equal(runtime.child.spawnfile,path.join(root,"node","node.exe"));
  assert.ok(runtime.child.spawnargs.includes(path.join(root,"python","python.exe")));
  const call=(id,url)=>runtime.invoke({protocol:1,id,method:"GET",path:url});
  const capabilities=await call("installed-fresh-capabilities","/api/v1/capabilities");
  assert.equal(capabilities.status,200); assert.equal(capabilities.body.success,true); assert.equal(capabilities.body.data.core_language,"typescript"); assert.equal(capabilities.body.data.setup_required,true);
  const me=await call("installed-fresh-auth-me","/api/v1/auth/me"); assert.equal(me.status,401);
  await runtime.stop(); assert.equal(runtime.child.exitCode,0); assert.equal(runtime.child.signalCode,null); assert.equal(helperFailed,false);
  step="readonly_schema";
  const db=openNativeDatabase(path.join(data,"marlin.sqlite3"),{readonly:true,fileMustExist:true}); let tables;
  try {
    db.defaultSafeIntegers();
    assert.equal(db.prepare("PRAGMA integrity_check").get().integrity_check,"ok"); assert.deepEqual(db.prepare("PRAGMA foreign_key_check").all(),[]);
    assert.equal(db.prepare("SELECT max(version) version FROM desktop_schema_versions").get().version,1n);
    assert.equal(db.prepare("SELECT count(*) n FROM sqlite_master WHERE name LIKE 'django_%'").get().n,0n);
    tables={};
    for(const name of ["users","knowledge_bases","knowledges","chunks","sessions","messages","models","model_usage","wiki_pages","device_sessions","desktop_jobs","task_records"]){ const count=db.prepare('SELECT count(*) n FROM "'+name+'"').get().n; assert.equal(count,0n); tables[name]=Number(count); }
  } finally { db.close(); }
  const result={format:"marlin-installed-native-ipc-v1",state:"verified",node:process.version,platform:process.platform,arch:process.arch,native:nativeResult,protocol:ready.protocol,runtime:ready.runtime,http_listener:false,capabilities_status:capabilities.status,auth_me_status:me.status,setup_required:true,schema_version:1,business_counts:tables,private_fork_closed_normally:true,fork_and_checks_ms:performance.now()-started,path_only_system32:true,paid_model_calls:0,renderer_gui_verified:false};
  fs.writeFileSync(output,JSON.stringify(result,null,2)+"\n",{flag:"wx"});
}
main().catch(async()=>{if(runtime){try{await runtime.stop();}catch{}}process.stderr.write(JSON.stringify({state:"failed",error_code:"installed_native_ipc_probe_failed",step})+"\n");process.exitCode=1;});
