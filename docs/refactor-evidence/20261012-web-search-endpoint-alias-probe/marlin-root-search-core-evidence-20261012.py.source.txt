from pathlib import Path
import datetime, hashlib, json, re

ROOT = Path('/home/liuxuedeng/Django-Agent')
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()

def summary(path):
    content = path.read_text()
    result = {}
    for name in ('tests', 'pass', 'fail', 'cancelled', 'skipped', 'todo', 'duration_ms'):
        matches = re.findall(r'^# ' + name + r' ([\d.]+)$', content, re.M)
        assert len(matches) == 1, (path.name, name)
        result[name] = float(matches[0]) if name == 'duration_ms' else int(matches[0])
    return {'path': path.relative_to(ROOT).as_posix(), 'sha256': sha(path), 'bytes': path.stat().st_size, **result}
