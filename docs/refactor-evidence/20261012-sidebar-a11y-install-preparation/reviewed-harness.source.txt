class Refusal(Exception):
    pass

def need(value, code):
    if not value:
        raise Refusal(code)

def digest(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()

def reap_owned_bootstrap(native, process_handle, job_handle, refusal=Refusal):
    """Observe only the existing owned handles for at most 30 seconds."""
    k, w = native[:2]; Accounting = native[3]
    started = time.monotonic(); deadline = started + 30
    observation = {"wait_result": None, "wait_last_error": 0, "owned_job_active_processes": None, "bootstrap_exit_code": None}
    polls = 0
    def result():
        return {**observation, "elapsed_ms": round((time.monotonic() - started) * 1000, 3), "budget_ms": 30000, "polls": polls, "reap_terminations": 0}
    def fail(code, last_error=0):
        print(json.dumps({"stage": "owned_bootstrap_reap_refused", "error_code": code, "api_last_error": last_error, **result()}, ensure_ascii=False), flush=True)
        raise refusal(code)
    while True:
        remaining = deadline - time.monotonic()
        if remaining <= 0: fail("bootstrap_reap_timeout")
        wait_ms = min(100, max(1, int(remaining * 1000)))
        wait = k.WaitForSingleObject(process_handle, wait_ms)
        polls += 1; observation["wait_result"] = int(wait)
        if wait == 0xffffffff:
            error = ctypes.get_last_error(); observation["wait_last_error"] = error
            fail("bootstrap_wait_failed", error)
        if wait not in (0, 258): fail("bootstrap_wait_result_invalid")
        accounting = Accounting()
        if not k.QueryInformationJobObject(job_handle, 1, ctypes.byref(accounting), ctypes.sizeof(accounting), None):
            fail("bootstrap_job_accounting_failed", ctypes.get_last_error())
        observation["owned_job_active_processes"] = int(accounting.active)
        if time.monotonic() > deadline: fail("bootstrap_reap_timeout")
        if wait == 0:
            code = w.DWORD()
            if not k.GetExitCodeProcess(process_handle, ctypes.byref(code)):
                fail("bootstrap_exit_code_read_failed", ctypes.get_last_error())
            observation["bootstrap_exit_code"] = int(code.value)
            if time.monotonic() > deadline: fail("bootstrap_reap_timeout")
            if code.value != 0: fail("bootstrap_exit_code_nonzero")
            if accounting.active == 0:
                value = result()
                print(json.dumps({"stage": "owned_bootstrap_reaped", **value}, ensure_ascii=False), flush=True)
                return value
            time.sleep(min(0.1, max(0, deadline - time.monotonic())))

def install_reap_adapter(module, relative, expected, function_name, prefix):
    """Replace precisely the reviewed tail guard; leave its owning loop intact."""
    archive = REPO / relative
    need(digest(archive) == expected, "reviewed_reap_archive_changed")
    tree = ast.parse(archive.read_text(encoding="utf-8"))
    selected = [node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == function_name]
    need(len(selected) == 1, "reviewed_reap_function_missing")
    function = copy.deepcopy(selected[0])
    original = ast.parse("need(k.WaitForSingleObject(process.process, 5000) == 0, '" + prefix + "_bootstrap_not_reaped')\ncode = w.DWORD()\nneed(k.GetExitCodeProcess(process.process, ctypes.byref(code)) and code.value == 0, '" + prefix + "_bootstrap_failed')").body
    replacement = ast.parse("_bootstrap_reap = reap_owned_bootstrap(native, process.process, job, Refusal)\ncode = w.DWORD(_bootstrap_reap['bootstrap_exit_code'])").body
    matches = returns = 0
    for node in ast.walk(function):
        for field, values in ast.iter_fields(node):
            if not isinstance(values, list): continue
            indices = [index for index in range(len(values) - 2) if all(ast.dump(values[index + offset], include_attributes=False) == ast.dump(original[offset], include_attributes=False) for offset in range(3))]
            for index in reversed(indices):
                values[index:index + 3] = copy.deepcopy(replacement); matches += 1
    for node in ast.walk(function):
        if isinstance(node, ast.Return):
            need(isinstance(node.value, ast.Dict), "reviewed_reap_return_changed")
            node.value.keys.append(ast.Constant(value="bootstrap_reap")); node.value.values.append(ast.Name(id="_bootstrap_reap", ctx=ast.Load())); returns += 1
    need(matches == 1 and returns == 1, "reviewed_reap_patch_count_changed")
    module.__dict__["reap_owned_bootstrap"] = reap_owned_bootstrap
    compiled = ast.Module(body=[function], type_ignores=[]); ast.fix_missing_locations(compiled)
    exec(compile(compiled, str(archive), "exec"), module.__dict__)
    return {"function": function_name, "archive_sha256": expected, "tail_replacements": matches, "return_fields_added": returns, "original_tail": ast.unparse(ast.Module(body=original, type_ignores=[])), "replacement_tail": ast.unparse(ast.Module(body=replacement, type_ignores=[])), "patched_function": ast.unparse(function)}

def module_from_archive(relative, expected, name):
    archive = REPO / relative
    need(digest(archive) == expected, "reviewed_archive_changed")
    module = ModuleType(name)
    module.__file__ = str(archive)
    exec(compile(archive.read_text(encoding="utf-8"), str(archive), "exec"), module.__dict__)
    return module

RACES = []
class CandidateExited(Exception):
    pass

def checked_identity_api(native, handle, pid, exe, operation, succeeded):
    if succeeded:
        return True
    error = ctypes.get_last_error()
    k, w = native[:2]
    wait = int(k.WaitForSingleObject(handle, 0))
    wait_error = ctypes.get_last_error() if wait == 4294967295 else 0
    code = w.DWORD()
    available = bool(k.GetExitCodeProcess(handle, ctypes.byref(code)))
    exit_error = ctypes.get_last_error() if not available else 0
    exited = wait == 0 and available
    RACES.append({'candidate_pid': int(pid), 'candidate_exe_basename': str(exe), 'api': operation, 'api_last_error': error, 'same_handle_wait_after': wait, 'wait_last_error': wait_error, 'exit_code_available': available, 'exit_code': int(code.value) if available else None, 'exit_code_last_error': exit_error, 'ignored_proven_exit': exited})
    if exited:
        raise CandidateExited()
    raise Refusal('native_candidate_identity_failed_not_proven_exited')

def confirm_candidate_alive(native, handle, pid, exe):
    # QueryPath/GetTimes can succeed after a process exits; check that same handle.
    k, w = native[:2]
    wait = int(k.WaitForSingleObject(handle, 0))
    if wait == 258:
        return
    wait_error = ctypes.get_last_error() if wait == 0xffffffff else 0
    code = w.DWORD(); available = bool(k.GetExitCodeProcess(handle, ctypes.byref(code)))
    exit_error = ctypes.get_last_error() if not available else 0
    exited = wait == 0 and available
    RACES.append({'candidate_pid': int(pid), 'candidate_exe_basename': str(exe), 'api': 'confirm_candidate_alive', 'api_last_error': 0, 'same_handle_wait_after': wait, 'wait_last_error': wait_error, 'exit_code_available': available, 'exit_code': int(code.value) if available else None, 'exit_code_last_error': exit_error, 'ignored_proven_exit': exited})
    if exited:
        raise CandidateExited()
    raise Refusal('native_candidate_liveness_not_proven')

def process_preflight(native, launch):
    """Native candidate snapshot; verify path/FILETIME/liveness on real handles.

    A process already exited between snapshot and OpenProcess is ignored only
    on ERROR_INVALID_PARAMETER; denied/unreadable candidate identity refuses.
    No CommandLine, CIM datetime conversion, signal or foreign handle termination.
    """
    k, w, *_ = native
    class Entry(ctypes.Structure):
        _fields_ = [("size", w.DWORD), ("usage", w.DWORD), ("pid", w.DWORD),
                    ("heap", ctypes.c_size_t), ("module", w.DWORD), ("threads", w.DWORD),
                    ("parent", w.DWORD), ("priority", ctypes.c_long), ("flags", w.DWORD),
                    ("exe", w.WCHAR * 260)]
    k.CreateToolhelp32Snapshot.argtypes, k.CreateToolhelp32Snapshot.restype = [w.DWORD, w.DWORD], w.HANDLE
    need(ctypes.sizeof(Entry) == 568, "native_process_entry_x64_layout_invalid")
    for name in ("Process32FirstW", "Process32NextW"):
        getattr(k, name).argtypes, getattr(k, name).restype = [w.HANDLE, ctypes.POINTER(Entry)], w.BOOL
    snapshot = k.CreateToolhelp32Snapshot(2, 0)
    need(snapshot not in (None, ctypes.c_void_p(-1).value), "native_process_snapshot_failed")
    entry = Entry(); entry.size = ctypes.sizeof(entry)
    try:
        present = k.Process32FirstW(snapshot, ctypes.byref(entry))
        need(present, "native_process_snapshot_empty")
        while present:
            if entry.exe.casefold() in {"marlin.exe", "node.exe", "python.exe", "pythonw.exe", "electron.exe"} or entry.pid == PREVIOUS_VISIBLE_PID:
                handle = k.OpenProcess(0x00101000, False, entry.pid)
                if not handle:
                    need(ctypes.get_last_error() == 87, "native_candidate_identity_unavailable")
                else:
                    try:
                        wait = k.WaitForSingleObject(handle, 0)
                        need(wait in (0, 258), "native_candidate_liveness_unavailable")
                        if wait == 258:
                            image = ctypes.create_unicode_buffer(32768); length = w.DWORD(len(image))
                            need(checked_identity_api(native, handle, int(entry.pid), entry.exe, 'QueryFullProcessImageNameW', k.QueryFullProcessImageNameW(handle, 0, image, ctypes.byref(length))), 'native_candidate_path_unavailable')
                            created, ended, kernel, user = (w.FILETIME() for _ in range(4))
                            need(checked_identity_api(native, handle, int(entry.pid), entry.exe, 'GetProcessTimes', k.GetProcessTimes(handle, ctypes.byref(created), ctypes.byref(ended), ctypes.byref(kernel), ctypes.byref(user))), 'native_candidate_creation_unavailable')
                            confirm_candidate_alive(native, handle, int(entry.pid), entry.exe)
                            filetime = (created.dwHighDateTime << 32) | created.dwLowDateTime
                            need(not (entry.pid == PREVIOUS_VISIBLE_PID and filetime == PREVIOUS_VISIBLE_CREATION), "previous_visible_requires_natural_tray_exit")
                            prefix = str(INSTALL).casefold()
                            need(entry.exe.casefold() != "marlin.exe" and not (image.value.casefold() == prefix or image.value.casefold().startswith(prefix + "\\")), "existing_installed_application_refused")
                    except CandidateExited:
                        pass
                    finally:
                        k.CloseHandle(handle)
            present = k.Process32NextW(snapshot, ctypes.byref(entry))
        need(ctypes.get_last_error() == 18, "native_process_enumeration_incomplete")
    finally:
        k.CloseHandle(snapshot)
    protected = launch.protected_handle(native)
    try:
        handle, identity = launch.process_handle(native, PROTECTED_PID)
        k.CloseHandle(handle)
    finally:
        k.CloseHandle(protected)
    return {"candidate_count": 0, "protected_outside_scope": [identity], "protected_count": 1,
            "related_processes": [], "historical_pids": [{"pid": PREVIOUS_VISIBLE_PID, "status": "exact_prior_generation_not_live"}],
            "command_lines_read": False, "native_filetime_checked": True, "nsis_powershell_policy_accepted": False}

    need(args.artifact_binding_sha256 and SHA.fullmatch(args.artifact_binding_sha256) and digest(binding_path) == args.artifact_binding_sha256, "reviewed_artifact_binding_hash_required")
    binding = install.load(binding_path)
    need(binding.get("state") == "verified_uninstalled" and binding.get("request_sha256") == args.binding_request_sha256 and binding.get("request") == req, "artifact_binding_scope_changed")
    identity, manifest, expected, audit = (binding[key] for key in ("identity", "manifest", "expected_installed_files", "asar_audit"))
    def formal_identity():
        need(digest(Path(req["freezePath"])) == req["sourceFreezeSHA"] and digest(STAGE / "prepared-build.json") == identity["prepared_sha256"] and digest(STAGE / "windows-gates.json") == identity["gates_sha256"], "bound_freeze_or_prepared_changed")
        need(digest(STAGE / "app/payload/source-manifest.json") == identity["payload_manifest_sha256"] and digest(STAGE / "asar-audit.json") == req["asarAuditSHA"], "bound_payload_or_audit_changed")
        need(ARTIFACT.stat().st_size == req["installerBytes"] and digest(ARTIFACT) == req["installerSHA"], "bound_installer_changed")
        return identity, manifest, expected
    def owned_scope():
        for path in (INSTALL, DATA, STAGE, ARTIFACT, folder): install.safe_path(path)
        launch.marker(OWNED / ".marlin-install-qa-owned", "marlin-ts-install-qa-20261010\n")
        need(digest(REPO / PRIOR_UPGRADE_REL) == PRIOR_UPGRADE_SHA, "prior_upgrade_changed")
        prior = install.load(REPO / PRIOR_UPGRADE_REL)
        need(prior.get("state") == "verified" and prior.get("scope", {}).get("installation_directory") == str(INSTALL) and prior["scope"].get("data_directory") == str(DATA) and prior["identity"].get("source_freeze_sha256") == PRIOR_INSTALL_FREEZE_SHA, "prior_owned_scope_changed")
        return {"marker_sha256": digest(OWNED / ".marlin-install-qa-owned"), "installation_directory": str(INSTALL), "data_directory": str(DATA), "guid": install.GUID, "old_receipt_sha256": PRIOR_UPGRADE_SHA}
    iteration = req["iteration"]
    install.REPO, install.EVIDENCE, install.STAGE, install.ARTIFACT = REPO, folder, STAGE, ARTIFACT
    install.OWNED, install.INSTALL, install.DATA = OWNED, INSTALL, DATA
    install.WORK = OWNED / (".sidebar-a11y-upgrade-20261012-" + str(iteration))
    install.SURVEY, install.PUBLIC_RECEIPT = folder / "readonly-baseline.json", folder / "executed-upgrade.json"
    install.OLD_RECEIPT, install.OLD_RECEIPT_SHA = REPO / PRIOR_UPGRADE_REL, PRIOR_UPGRADE_SHA
    install.INSTALLER_SHA, install.INSTALLER_SIZE = req["installerSHA"], req["installerBytes"]
    install.__file__ = str(Path(__file__))
    install.formal_identity, install.owned_scope = formal_identity, owned_scope
    install.process_snapshot = lambda native: installer_preflight(native, install, launch)
    launch.REPO, launch.EVIDENCE, launch.INSTALL, launch.RUNTIME, launch.DATA = REPO, folder, INSTALL, INSTALL / "resources/runtime", DATA
    launch.WORK = OWNED / (".sidebar-a11y-launch-20261012-" + str(iteration))
    launch.SMOKE_DATA = OWNED / ("启动验收 侧栏可访问性 20261012 第" + str(iteration) + "轮")
    launch.IPC_DATA = launch.WORK / "私有 IPC 空数据"
    launch.__file__ = str(Path(__file__))
    launch.process_preflight = lambda: process_preflight(launch.native_api(), launch)
    def installed_identity():
        formal_identity()
        upgrade = install.load(install.PUBLIC_RECEIPT)
        need(digest(install.PUBLIC_RECEIPT) == args.upgrade_receipt_sha256 and upgrade.get("state") == "verified" and upgrade.get("identity") == identity and upgrade.get("scope") == owned_scope(), "verified_bound_upgrade_required")
        need(upgrade["installed"].get("full_code_inventory_matches") is True and upgrade["installed"].get("full_runtime_inventory_matches") is True, "full_installed_verification_required")
        need(digest(INSTALL / "resources/app.asar") == identity["asar_sha256"] and digest(launch.RUNTIME / "source-manifest.json") == identity["payload_manifest_sha256"], "installed_asar_or_manifest_changed")
        critical = {name: sha for name, sha in expected.items() if "/" not in name}
        runtime = manifest["files"]
        python_keys = {"python.exe", "pythonw.exe", "python313.dll", "python3.dll", "python313._pth", "python313.zip", "vcruntime140.dll", "vcruntime140_1.dll", "ucrtbase.dll"}
        selected = {name: sha for name, sha in runtime.items() if name.startswith(("core/dist/", "frontend/", "workers/")) or name == "core/package.json" or name.startswith("node/") and name.count("/") == 1 or name.startswith("python/") and name.removeprefix("python/") in python_keys or name.startswith("core/") and name.endswith((".node", ".dll"))}
        for name, sha in critical.items():
            path = INSTALL / name; install.safe_path(path); need(digest(path) == sha, "installed_critical_file_changed")
        for name, sha in selected.items():
            path = launch.RUNTIME / name; install.safe_path(path); need(digest(path) == sha, "installed_critical_runtime_changed")
        need(runtime.get("node/node.exe") == launch.NODE_SHA and runtime.get("python/python.exe") == launch.PYTHON_SHA, "bundled_executable_pin_changed")
        uninstaller = INSTALL / "Uninstall Marlin.exe"; install.safe_path(uninstaller)
        need(digest(uninstaller) == upgrade["installed"]["uninstaller_sha256"], "installed_uninstaller_changed")
        assets = launch.read_asar_assets(INSTALL / "resources/app.asar", audit)
        return {**identity, "executables_sha256": {"Marlin.exe": critical["Marlin.exe"], "node/node.exe": launch.NODE_SHA, "python/python.exe": launch.PYTHON_SHA}, "critical_install_files_checked": len(critical) + 3, "critical_runtime_files_checked": len(selected), "whole_inventory_rehashed": False}, assets
    launch.installed_identity = installed_identity
    original_hidden = launch.check_hidden_result
    def check_hidden_result(native, report, assets):
        original_hidden(native, report, assets)
        result = install.load(launch.SMOKE_DATA / "smoke-result.json")
        need(result.get("pageReady") is True and result.get("pageReadyKind") == "login" and result.get("startup_interval") == "runtime_start_to_renderer_readiness_result", "real_login_page_readiness_not_verified")
        for key in ("mounted_ms", "page_ready_ms", "renderer_wait_ms", "startup_ms"):
            need(type(result.get(key)) in (int, float) and result[key] >= 0, "readiness_measurement_missing")
        need(result["mounted_ms"] <= result["page_ready_ms"] <= result["renderer_wait_ms"], "readiness_measurement_order_invalid")
        report["hidden_smoke"].update(page_ready=True, page_ready_kind="login", mount_only=False, page_ready_not_verified=False, mounted_ms=result["mounted_ms"], page_ready_ms=result["page_ready_ms"], renderer_wait_ms=result["renderer_wait_ms"], startup_interval=result["startup_interval"], screenshot_requires_root_visual_review=True, manual_gui_verified=False)
    launch.check_hidden_result = check_hidden_result

def execute(args):
    install = module_from_archive(INSTALL_ARCHIVE_REL, INSTALL_ARCHIVE_SHA, "marlin_integrated_install_archive")
    launch = module_from_archive(LAUNCH_ARCHIVE_REL, LAUNCH_ARCHIVE_SHA, "marlin_integrated_launch_archive")
    install.native_required()
    need(digest(Path(__file__)) == args.reviewed_script_sha256, "reviewed_script_changed")
    install_reap_adapter(install, INSTALL_ARCHIVE_REL, INSTALL_ARCHIVE_SHA, "run_installer_tree", "installer")
    install_reap_adapter(launch, LAUNCH_ARCHIVE_REL, LAUNCH_ARCHIVE_SHA, "run_owned_tree", "owned")
    req = request(args, install)
    folder = EVIDENCE / ("iteration-" + str(req["iteration"]))
    install.safe_path(folder)
    if args.phase == "bind-artifact":
        need(not folder.exists(), "iteration_already_bound_manual_review_required")
        folder.mkdir()
        try:
            return artifact_binding(args, install, launch, req, folder)
        except BaseException as error:
            install.write_new(folder / "artifact-binding-failed.json", {"state": "failed_manual_review_required", "request_sha256": args.binding_request_sha256, "source_freeze_sha256": req["sourceFreezeSHA"], "error_code": str(error) if isinstance(error, Refusal) or type(error).__name__ == "Refusal" else "safe_binding_failure_details_withheld", "installer_executed": False, "application_started": False, "automatic_retry_or_resume": False})
            raise
    need(folder.is_dir(), "bound_iteration_missing")
    configure(args, install, launch, req, folder)
    if args.phase == "baseline":
        return install.survey_readonly()
    if args.phase == "upgrade":
        need(args.baseline_sha256 and SHA.fullmatch(args.baseline_sha256), "reviewed_current_baseline_hash_required")
        args.readonly_survey_sha256 = args.baseline_sha256
        return install.execute_upgrade(args)
    need(args.upgrade_receipt_sha256 and SHA.fullmatch(args.upgrade_receipt_sha256), "bound_upgrade_receipt_required")
    launch.UPGRADE_SHA = args.upgrade_receipt_sha256
    launch.RETAINED_BASELINE = install.WORK / "after.json"
    upgrade = install.load(install.PUBLIC_RECEIPT)
    launch.RETAINED_BASELINE_SHA = upgrade["retained_after_sha256"]
    need(args.phase == "hidden-smoke" and args.hidden_receipt_sha256 is None or args.phase == "visible-retained" and args.hidden_receipt_sha256 and SHA.fullmatch(args.hidden_receipt_sha256), "exact_preceding_hidden_receipt_required")
    return launch.execute(args)

def main():
    for stream in (sys.stdout, sys.stderr):
        if hasattr(stream, "reconfigure"):
            stream.reconfigure(encoding="utf-8")
    parser = argparse.ArgumentParser(description=__doc__, allow_abbrev=False)
    parser.add_argument("--phase", choices=PHASES)
    parser.add_argument("--iteration", type=int)
    parser.add_argument("--execute-reviewed", action="store_true")
    for field in ("reviewed-script", "binding-request", "source-freeze", "artifact-binding", "baseline", "upgrade-receipt", "hidden-receipt"):
        parser.add_argument("--" + field + "-sha256")
    flags = [token for token in sys.argv[1:] if token.startswith("--")]
    need(len(flags) == len(set(flags)) and not any("=" in flag for flag in flags), "duplicate_or_equals_flags_refused")
    args = parser.parse_args()
    if not args.execute_reviewed:
        need(args.iteration is None, "iteration_requires_explicit_execution")
        need(not any(value for key, value in vars(args).items() if key.endswith("sha256")), "hashes_without_explicit_execution_refused")
        print(json.dumps({"state": "review_only", "phase": args.phase, "supported_phases": PHASES, "script_sha256": digest(Path(__file__)), "artifact_binding": "unbound_until_reviewed_request_and_real_success_receipts", "fixed_installation": str(INSTALL), "fixed_retained_data": str(DATA), "application_started": False, "installer_executed": False, "windows_process_operations": 0, "registry_writes": False, "data_changes": False, "paid_model_calls": 0}, ensure_ascii=False))
        return
    need(args.phase in PHASES and args.reviewed_script_sha256 and SHA.fullmatch(args.reviewed_script_sha256), "explicit_phase_and_reviewed_script_hash_required")
    need(args.binding_request_sha256 and SHA.fullmatch(args.binding_request_sha256) and args.source_freeze_sha256 and SHA.fullmatch(args.source_freeze_sha256), "explicit_reviewed_binding_and_actual_freeze_required")
    need(type(args.iteration) is int and 1 <= args.iteration <= 99, "explicit_iteration_required")
    phase_hashes = {"bind-artifact": (), "baseline": ("artifact_binding_sha256",), "upgrade": ("artifact_binding_sha256", "baseline_sha256"), "hidden-smoke": ("artifact_binding_sha256", "upgrade_receipt_sha256"), "visible-retained": ("artifact_binding_sha256", "upgrade_receipt_sha256", "hidden_receipt_sha256")}
    for field in ("artifact_binding_sha256", "baseline_sha256", "upgrade_receipt_sha256", "hidden_receipt_sha256"):
        value = getattr(args, field)
        need(value and SHA.fullmatch(value) if field in phase_hashes[args.phase] else value is None, "exact_phase_receipt_hashes_required")
    result = execute(args)
    result["same_handle_identity_observations"] = RACES
    print(json.dumps(result, ensure_ascii=False), flush=True)

if __name__ == "__main__":
    try:
        main()
    except BaseException as error:
        if isinstance(error, SystemExit): raise
        print(json.dumps({"state": "refused_or_failed", "error_code": str(error) if isinstance(error, Refusal) or type(error).__name__ == "Refusal" else "safe_failure_details_withheld", "automatic_retry_or_resume": False, "paid_model_calls": 0, "same_handle_identity_observations": RACES}), flush=True)
        sys.exit(1)
