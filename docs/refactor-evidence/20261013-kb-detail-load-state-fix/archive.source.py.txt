from pathlib import Path
import hashlib
import json
import subprocess

root = Path('/home/liuxuedeng/Django-Agent')
evidence = root / 'docs/refactor-evidence/20261013-kb-detail-load-state-fix'
owned = [root / 'frontend/src/views/KnowledgeDetail.vue', root / 'frontend/src/views/KnowledgeDetail.load-state.test.mjs']
before = Path('/tmp/kb-detail-load-state-fix/KnowledgeDetail.before.vue')
helper = Path(__file__)
digest = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
expected = (evidence / 'before.sha256').read_text().split()[0]
assert digest(before) == expected, 'before-copy SHA mismatch; do not remove'
diff = subprocess.run(['git', 'diff', '--', str(owned[0].relative_to(root))], cwd=root, text=True, capture_output=True, check=True).stdout
added = subprocess.run(['git', 'diff', '--no-index', '--', '/dev/null', str(owned[1].relative_to(root))], cwd=root, text=True, capture_output=True)
assert added.returncode == 1 and added.stdout.startswith('diff --git ')
(evidence / 'code-complete.diff').write_text(diff + added.stdout)
(evidence / 'archive.source.py.txt').write_text(helper.read_text())
cleanup = []
for path in [before, helper]:
    cleanup.append({'path': str(path), 'sha256': digest(path), 'removed': False})
for entry in cleanup:
    path = Path(entry['path'])
    assert digest(path) == entry['sha256'], 'temporary file changed; do not remove'
    path.unlink()
    entry['removed'] = not path.exists()
helper.parent.rmdir()
(evidence / 'temporary-cleanup.json').write_text(json.dumps({'files': cleanup, 'directory_removed': not helper.parent.exists()}, indent=2) + '\n')
