def binding(path):
    p = Path(path)
    if not p.is_absolute(): p = ROOT / p
    raw = p.read_bytes()
    return {'path': str(p.relative_to(ROOT)), 'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}

def read(path): return json.loads((ROOT / 'docs/refactor-evidence' / path).read_text())

def cmd(argv):
    v = subprocess.run(argv, cwd=ROOT, capture_output=True)
    assert v.returncode == 0, (argv, v.returncode, v.stderr.decode())
    return v

diff = []
new_links = []
def links(text):
    return [t.strip('<>') for _, t in re.findall(r'\[([^\]\n]+)\]\((<[^>]+>|[^)\n]+)\)', text)
            if not urlsplit(t.strip('<>')).scheme and not t.startswith('#')]
for item in BEFORE['documents']:
    path = item['path']
    old = (OUT / ('before-' + Path(path).name + '.source.txt')).read_text()
    assert hashlib.sha256(old.encode()).hexdigest() == item['sha256']
    new = (ROOT / path).read_text()
    assert old != new
    diff.extend(difflib.unified_diff(old.splitlines(True), new.splitlines(True), fromfile='before/' + path, tofile=path))
    for target, count in (Counter(links(new)) - Counter(links(old))).items():
        new_links.extend([{'source': path, 'target': target}] * count)
(OUT / 'four-owned-documents.diff').write_text(''.join(diff))
report_path = str((OUT / 'REPORT.md').relative_to(ROOT))
