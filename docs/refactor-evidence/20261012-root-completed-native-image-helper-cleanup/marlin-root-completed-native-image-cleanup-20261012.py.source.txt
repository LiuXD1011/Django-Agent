]
assert len(names) == len(set(names))
assert not EVIDENCE.exists()
EVIDENCE.mkdir()
rows = []
for name in names:
 source = Path('/tmp') / name
 assert source.is_file() and not source.is_symlink(), f'exact completed helper missing: {name}'
 contents = source.read_bytes()
 target = EVIDENCE / (name + '.source.txt')
 target.write_bytes(contents)
 digest = hashlib.sha256(contents).hexdigest()
 assert hashlib.sha256(target.read_bytes()).hexdigest() == digest
 rows.append({'source': str(source), 'archive': str(target.relative_to(ROOT)), 'sha256': digest, 'bytes': len(contents)})
for row in rows:
 source = Path(row['source'])
 assert hashlib.sha256(source.read_bytes()).hexdigest() == row['sha256']
 source.unlink()
 assert not source.exists()
value = {'state': 'verified_exact_completed_helpers_archived_and_deleted', 'createdUtc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'rows': rows, 'count': len(rows), 'activeBuildBrowserAgentAndAuthenticationHelpersExcluded': True, 'businessFilesDeleted': 0, 'processSignals': 0}
(EVIDENCE / 'receipt.json').write_text(json.dumps(value, ensure_ascii=False, indent=2), encoding='utf8')
print(json.dumps({key: value[key] for key in ('state', 'count', 'businessFilesDeleted', 'processSignals')}))
