def binding(path):
    p = Path(path)
    if not p.is_absolute():
        p = ROOT / p
    data = p.read_bytes()
    return {'path': str(p.relative_to(ROOT)) if p.is_relative_to(ROOT) else str(p),
            'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def verify(item):
    actual = binding(item['path'])
    assert actual['sha256'] == item['sha256'], item['path']
    assert actual['bytes'] == item.get('bytes', item.get('size')), item['path']
    return actual

def read(path):
    return json.loads((ROOT / path).read_text())

def command(argv):
    r = subprocess.run(argv, cwd=ROOT, capture_output=True)
    assert r.returncode == 0, (argv, r.returncode, r.stderr.decode())
    return {'command': argv, 'actual_exit_code': r.returncode,
            'stdout': r.stdout.decode(), 'stderr': r.stderr.decode()}

links = []
link_files = documents + [str((OUT / 'PLAN.md').relative_to(ROOT))]
for path in link_files:
    p = ROOT / path
    for label, target in re.findall(r'\[([^\]\n]+)\]\((<[^>]+>|[^)\n]+)\)', p.read_text()):
        target = target.strip('<>')
        if urlsplit(target).scheme or target.startswith('#'):
            continue
        clean = unquote(target.split('#', 1)[0].split('?', 1)[0])
        clean = re.sub(r':\d+$', '', clean)
        resolved = (ROOT / clean) if clean.startswith('/') is False and clean.startswith('docs/') else (p.parent / clean)
        if clean.startswith('/'):
            resolved = Path(clean)
        resolved = resolved.resolve()
        assert resolved.exists(), (path, target)
        links.append({'source': path, 'target': target, 'exists': True})
