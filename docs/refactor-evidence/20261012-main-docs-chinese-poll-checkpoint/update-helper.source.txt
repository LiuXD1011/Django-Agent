def paragraph(name, starts, replacement):
    parts = texts[name].split('\n\n')
    matches = [i for i, part in enumerate(parts) if part.startswith(starts)]
    assert len(matches) == 1, (name, starts, len(matches))
    parts[matches[0]] = replacement
    texts[name] = '\n\n'.join(parts)
def exact(name, old, new):
    assert texts[name].count(old) == 1, (name, old)
    texts[name] = texts[name].replace(old, new)
def row(name, identifier, replacement):
    lines = texts[name].splitlines()
    matches = [i for i, line in enumerate(lines) if line.startswith('| ' + identifier + ' |')]
    assert len(matches) == 1, identifier
    lines[matches[0]] = replacement
    texts[name] = '\n'.join(lines) + '\n'
