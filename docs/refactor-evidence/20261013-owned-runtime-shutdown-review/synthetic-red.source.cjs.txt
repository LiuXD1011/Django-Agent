const tick=()=>Promise.resolve().then(()=>Promise.resolve());
test('baseline: graceful IPC exit keeps one stop Promise and no PID helper',async()=>{

test('RED: after timeout helper creation, delayed PID lookup must not reach a different generation',async()=>{

test('RED: stop must await its timeout helper terminal state when the core exits first',async()=>{
