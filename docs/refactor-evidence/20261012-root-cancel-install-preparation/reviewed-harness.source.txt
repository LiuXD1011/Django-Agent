def need(value, code):
    if not value:

def digest(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def json_hash(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def safe_path(path):
    for candidate in (Path(path), *Path(path).parents):
        try:
            info = candidate.lstat()
        except FileNotFoundError:
            continue
        need(not stat.S_ISLNK(info.st_mode) and not getattr(info, "st_file_attributes", 0) & 0x400, "reparse_path_rejected")


def inventory(root):
    safe_path(root)
    need(Path(root).is_dir(), "inventory_directory_missing")
    result = {}

    def visit(folder):
        with os.scandir(folder) as entries:
            for entry in sorted(entries, key=lambda item: item.name):
                info = entry.stat(follow_symlinks=False)
                need(not stat.S_ISLNK(info.st_mode) and not getattr(info, "st_file_attributes", 0) & 0x400, "reparse_inventory_rejected")
                file = Path(entry.path)
                if stat.S_ISDIR(info.st_mode):
                    visit(file)
                else:
                    need(stat.S_ISREG(info.st_mode), "nonregular_inventory_file")
                    result[file.relative_to(root).as_posix()] = digest(file)

    visit(Path(root))
    return result

def native_api():
    from ctypes import wintypes as w
    k = ctypes.WinDLL("kernel32", use_last_error=True)
    size = ctypes.c_size_t

    class Basic(ctypes.Structure):
        _fields_ = [("user", ctypes.c_longlong), ("job", ctypes.c_longlong), ("flags", w.DWORD), ("min", size), ("max", size), ("limit", w.DWORD), ("affinity", size), ("priority", w.DWORD), ("scheduling", w.DWORD)]

    class Io(ctypes.Structure):
        _fields_ = [(name, ctypes.c_ulonglong) for name in ("reads", "writes", "other", "read_bytes", "write_bytes", "other_bytes")]

    class Extended(ctypes.Structure):
        _fields_ = [("basic", Basic), ("io", Io)] + [(name, size) for name in ("process", "job", "peak_process", "peak_job")]

    class Accounting(ctypes.Structure):
        _fields_ = [(name, ctypes.c_longlong) for name in ("user", "kernel", "period_user", "period_kernel")] + [(name, w.DWORD) for name in ("faults", "total", "active", "terminated")]

    class Startup(ctypes.Structure):
        _fields_ = [("cb", w.DWORD), ("reserved", w.LPWSTR), ("desktop", w.LPWSTR), ("title", w.LPWSTR)] + [(name, w.DWORD) for name in ("x", "y", "width", "height", "columns", "rows", "fill", "flags")] + [("show", w.WORD), ("reserved_size", w.WORD), ("reserved_bytes", ctypes.c_void_p), ("stdin", w.HANDLE), ("stdout", w.HANDLE), ("stderr", w.HANDLE)]

    class Process(ctypes.Structure):
        _fields_ = [("process", w.HANDLE), ("thread", w.HANDLE), ("pid", w.DWORD), ("tid", w.DWORD)]

    signatures = {
        "CreateJobObjectW": ([ctypes.c_void_p, w.LPCWSTR], w.HANDLE),
        "SetInformationJobObject": ([w.HANDLE, ctypes.c_int, ctypes.c_void_p, w.DWORD], w.BOOL),
        "QueryInformationJobObject": ([w.HANDLE, ctypes.c_int, ctypes.c_void_p, w.DWORD, ctypes.c_void_p], w.BOOL),
        "AssignProcessToJobObject": ([w.HANDLE, w.HANDLE], w.BOOL),
        "TerminateJobObject": ([w.HANDLE, w.UINT], w.BOOL),
        "TerminateProcess": ([w.HANDLE, w.UINT], w.BOOL),
        "CreateProcessW": ([w.LPCWSTR, w.LPWSTR, ctypes.c_void_p, ctypes.c_void_p, w.BOOL, w.DWORD, ctypes.c_void_p, w.LPCWSTR, ctypes.POINTER(Startup), ctypes.POINTER(Process)], w.BOOL),
        "ResumeThread": ([w.HANDLE], w.DWORD),
        "WaitForSingleObject": ([w.HANDLE, w.DWORD], w.DWORD),
        "GetExitCodeProcess": ([w.HANDLE, ctypes.POINTER(w.DWORD)], w.BOOL),
        "CloseHandle": ([w.HANDLE], w.BOOL),
        "OpenProcess": ([w.DWORD, w.BOOL, w.DWORD], w.HANDLE),
        "QueryFullProcessImageNameW": ([w.HANDLE, w.DWORD, w.LPWSTR, ctypes.POINTER(w.DWORD)], w.BOOL),
        "GetProcessTimes": ([w.HANDLE, ctypes.POINTER(w.FILETIME), ctypes.POINTER(w.FILETIME), ctypes.POINTER(w.FILETIME), ctypes.POINTER(w.FILETIME)], w.BOOL),
        "CreateFileW": ([w.LPCWSTR, w.DWORD, w.DWORD, ctypes.c_void_p, w.DWORD, w.DWORD, w.HANDLE], w.HANDLE),
    }
    for name, (arguments, result) in signatures.items():
        getattr(k, name).argtypes, getattr(k, name).restype = arguments, result
    return k, w, Extended, Accounting, Startup, Process


def process_handle(native, pid):
    k, w, *_ = native
    handle = k.OpenProcess(0x00101000, False, pid)
    need(handle, "protected_process_unavailable")
    try:
        image = ctypes.create_unicode_buffer(32768)
        length = w.DWORD(len(image))
        need(k.QueryFullProcessImageNameW(handle, 0, image, ctypes.byref(length)), "protected_process_path_unavailable")
        creation, exit_time, kernel, user = (w.FILETIME() for _ in range(4))
        need(k.GetProcessTimes(handle, ctypes.byref(creation), ctypes.byref(exit_time), ctypes.byref(kernel), ctypes.byref(user)), "protected_process_creation_unavailable")
        need(k.WaitForSingleObject(handle, 0) == 258, "protected_process_already_exited")
        identity = {"pid": pid, "image": image.value, "creation_filetime": (creation.dwHighDateTime << 32) | creation.dwLowDateTime}
        return handle, identity
    except BaseException:
        k.CloseHandle(handle)
        raise

def registry_state():
    import winreg
    rows = []
    suffixes = ("Software\\" + GUID, "Software\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\" + GUID)
    for view, view_name in ((winreg.KEY_WOW64_64KEY, "64"), (winreg.KEY_WOW64_32KEY, "32")):
        for hive, name in ((winreg.HKEY_LOCAL_MACHINE, "HKLM"), (winreg.HKEY_CURRENT_USER, "HKCU")):
            for index, suffix in enumerate(suffixes):
                row = {"hive": name, "view": view_name, "key": suffix, "exists": False}
                try:
                    with winreg.OpenKey(hive, suffix, 0, winreg.KEY_READ | view) as key:
                        need(name == "HKCU", "machine_installation_refused")
                        if index == 0:
                            value = winreg.QueryValueEx(key, "InstallLocation")[0]
                            need(isinstance(value, str) and value.casefold() == str(INSTALL).casefold(), "foreign_install_location")
                            row["install_location"] = value
                        else:
                            need(winreg.QueryValueEx(key, "UninstallString")[0] == '"' + str(INSTALL / "Uninstall Marlin.exe") + '" /currentuser', "foreign_uninstaller_registration")
                            need(winreg.QueryValueEx(key, "QuietUninstallString")[0] == '"' + str(INSTALL / "Uninstall Marlin.exe") + '" /currentuser /S', "foreign_quiet_uninstaller_registration")
                            row["uninstall_executable"] = str(INSTALL / "Uninstall Marlin.exe")
                        row["exists"] = True
                except FileNotFoundError:
                    need(name == "HKLM", "owned_registration_missing")
                rows.append(row)
    return rows

def retained_state():
    # The caller holds deny-write/delete access for this offline database.
    # immutable=1 also avoids creating/modifying WAL or SHM from our RO read.
    database = DATA / "marlin.sqlite3"
    connection = sqlite3.connect(database.as_uri() + "?mode=ro&immutable=1", uri=True)
    try:
        connection.execute("PRAGMA query_only=ON")
        need(connection.execute("PRAGMA integrity_check").fetchone()[0] == "ok", "database_integrity_failed")
        need(not connection.execute("PRAGMA foreign_key_check").fetchall(), "database_foreign_keys_failed")
        rows = {}
        for table in TABLES:
            # Private row values remain in memory only; all output is counts
            # and hashes. Python integers preserve native SQLite 64-bit IDs.
            values = list(connection.execute('SELECT * FROM "' + table + '" ORDER BY id'))
            rows[table] = {"count": len(values), "rows_sha256": json_hash(values)}
        user = connection.execute("SELECT tenant_id FROM users WHERE id=? AND is_active=1 AND deleted_at IS NULL", (USER,)).fetchone()
        kb = connection.execute("SELECT tenant_id FROM knowledge_bases WHERE id=? AND deleted_at IS NULL", (KB,)).fetchone()
        need(user and kb and str(user[0]) == str(kb[0]), "qa_user_or_kb_scope_invalid")
    finally:
        connection.close()
    credentials = inventory(DATA / ".secrets")
    need(credentials, "credential_hash_material_missing")
    need(all(rows[name]["count"] == 0 for name in ("models", "model_usage", "desktop_jobs", "task_records")), "unexpected_models_usage_or_jobs")
    return {"business_tables": rows, "credential_material_sha256": credentials, "data_files_sha256": inventory(DATA)}

        limits = Extended()
        limits.basic.flags = 0x2000
        need(k.SetInformationJobObject(job, 9, ctypes.byref(limits), ctypes.sizeof(limits)), "private_job_limits_failed")
        with log.open("xb") as output, open(os.devnull, "rb") as input_stream:
            handles = [msvcrt.get_osfhandle(stream.fileno()) for stream in (input_stream, output)]
            for handle in handles:
                os.set_handle_inheritable(handle, True)
            try:
                startup = Startup()
                startup.cb = ctypes.sizeof(startup)
                startup.flags = 0x100
                startup.stdin, startup.stdout, startup.stderr = handles[0], handles[1], handles[1]
                environment = ctypes.create_unicode_buffer("\0".join(key + "=" + value for key, value in sorted(minimal_env().items(), key=lambda item: item[0].casefold())) + "\0\0")
                need(k.CreateProcessW(str(ARTIFACT), ctypes.create_unicode_buffer(command), None, None, True, 0x4 | 0x400 | 0x08000000, environment, str(OWNED), ctypes.byref(startup), ctypes.byref(process)), "private_installer_create_failed")
            finally:
                for handle in handles:
                    os.set_handle_inheritable(handle, False)
            need(k.AssignProcessToJobObject(job, process.process), "private_installer_assign_failed")
            assigned = True
            need(k.ResumeThread(process.thread) != 0xFFFFFFFF, "private_installer_resume_failed")
            deadline = started + 900
            while True:
                accounting = Accounting()
                need(k.QueryInformationJobObject(job, 1, ctypes.byref(accounting), ctypes.sizeof(accounting), None), "private_job_query_failed")
                if accounting.active == 0:
                    completed = True
                    break
                need(time.monotonic() < deadline, "owned_installer_timeout")
                time.sleep(0.03)
            need(k.WaitForSingleObject(process.process, 5000) == 0, "installer_bootstrap_not_reaped")
            code = w.DWORD()
            need(k.GetExitCodeProcess(process.process, ctypes.byref(code)) and code.value == 0, "installer_bootstrap_failed")
        return {"bootstrap_exit_code": code.value, "owned_job_active_processes": 0, "complete_owned_tree_exited": True, "descendant_exit_codes_claimed": False, "duration_ms": round((time.monotonic() - started) * 1000, 3), "log_sha256": digest(log), "command": command, "private_job_timeout_seconds": 900}
    finally:
        if process.process and not assigned:
            k.TerminateProcess(process.process, 125)
            k.WaitForSingleObject(process.process, 5000)
        elif assigned and not completed:
            k.TerminateJobObject(job, 124)
            deadline = time.monotonic() + 5
            while time.monotonic() < deadline:
                accounting = Accounting()
                if k.QueryInformationJobObject(job, 1, ctypes.byref(accounting), ctypes.sizeof(accounting), None) and accounting.active == 0:
                    break
                time.sleep(0.03)
        k.CloseHandle(job)
        for handle in (process.thread, process.process):
            if handle:
                k.CloseHandle(handle)


def native_required():
    need(os.name == "nt" and platform.machine().lower() in {"amd64", "x86_64"} and sys.maxsize > 2**32, "native_windows_x64_required")
    need(sys.version_info[:2] == (3, 13), "test_host_python313_required")
    sys.dont_write_bytecode = True


def survey_readonly():
    native_required()
    need(not SURVEY.exists(), "survey_already_exists_review_required")
    scope = owned_scope()
    native = native_api()
    process = process_snapshot(native)
    registration = registry_state()
    lock = database_guard(native)
    try:
        before = retained_state()
        old_identity = previous_installed_identity()
        after = retained_state()
        need(before == after, "data_changed_during_readonly_survey")
    finally:
        native[0].CloseHandle(lock)
    current = process_snapshot(native)
    need(current["protected_outside_scope"] == process["protected_outside_scope"], "process_identity_changed_during_survey")
    record = {"format": "marlin-cancel-install-readonly-survey-v1", "state": "verified_readonly", "created_utc": datetime.now(timezone.utc).isoformat(), "scope": scope, "processes": process, "registration": registration, "existing_installation": old_identity, "retained": before, "data_read_method": "Win32 deny-write/delete handle + SQLite immutable readonly, zero WAL", "application_started": False, "installer_executed": False, "registry_writes": False, "data_changes": False, "row_values_or_credentials_reported": False, "script_sha256_at_survey": digest(Path(__file__))}
    write_new(SURVEY, record)
    return {"state": record["state"], "receipt": str(SURVEY), "receipt_sha256": digest(SURVEY), "historical_pids": process["historical_pids"], "protected_processes": process["protected_outside_scope"], "installed_files": old_identity["installed_files"], "data_files": len(before["data_files_sha256"]), "business_counts": {name: row["count"] for name, row in before["business_tables"].items()}, "installer_executed": False, "registry_writes": False, "data_changes": False}


def execute_upgrade(args):
    native_required()
    need(digest(Path(__file__)) == args.reviewed_script_sha256, "reviewed_script_hash_changed")
    need(digest(SURVEY) == args.readonly_survey_sha256, "reviewed_survey_hash_changed")
    survey = load(SURVEY)
    need(survey.get("state") == "verified_readonly" and survey.get("scope") == owned_scope(), "reviewed_survey_scope_changed")
    need(not WORK.exists() and not PUBLIC_RECEIPT.exists(), "upgrade_already_attempted_review_required")
    identity, manifest, expected = formal_identity()
    native = native_api()
    k = native[0]
    process = process_snapshot(native)
    need(process["protected_outside_scope"] == survey["processes"]["protected_outside_scope"], "protected_process_survey_changed")
    need(registry_state() == survey["registration"], "scoped_registration_changed")
    need(previous_installed_identity() == survey["existing_installation"], "old_installed_identity_changed")
    protected = []
    lock = None
    try:
        for entry in process["protected_outside_scope"]:
            handle, actual = process_handle(native, entry["pid"])
            protected.append(handle)
            need(actual == entry, "protected_process_creation_changed")
        lock = database_guard(native)
        before = retained_state()
        need(before == survey["retained"], "retained_survey_baseline_changed")
        safe_path(WORK)
        WORK.mkdir()
        write_new(WORK / "attempt.json", {"format": "marlin-cancel-installed-upgrade-attempt-v1", "identity": identity, "reviewed_script_sha256": args.reviewed_script_sha256, "readonly_survey_sha256": args.readonly_survey_sha256, "automatic_retry_or_resume": False})
        write_new(WORK / "before.json", before)
        report = {"format": "marlin-cancel-installed-code-upgrade-v1", "state": "running", "created_utc": datetime.now(timezone.utc).isoformat(), "identity": identity, "scope": survey["scope"], "reviewed_script_sha256": args.reviewed_script_sha256, "readonly_survey_sha256": args.readonly_survey_sha256, "same_desktop_version_code_upgrade": True, "version_migration_claimed": False, "application_started": False, "manual_gui_verified": False, "api_calls": 0, "paid_model_calls": 0, "registry_writes_by_harness": False, "old_core_started": False, "generic_process_kill_used": False, "delete_app_data_requested": False, "automatic_retry_or_resume": False}
        try:
            # Recheck process absence immediately before creating NSIS. Parent
            # must keep this fixed installation quiescent throughout execution.
            process_snapshot(native)
            need(owned_scope() == survey["scope"] and registry_state() == survey["registration"], "owned_scope_changed_at_installer_launch")
            report["installer_execution"] = run_installer_tree(native)
            process_snapshot(native)
            need(registry_state() == survey["registration"], "post_upgrade_registry_scope_changed")
            report["installed"] = installed_identity(identity, manifest, expected)
            after = retained_state()
            need(after == before, "upgrade_changed_business_files_or_credentials")
            write_new(WORK / "after.json", after)
            need(formal_identity()[0] == identity, "current_source_or_payload_changed_during_upgrade")
            need(all(k.WaitForSingleObject(handle, 0) == 258 for handle in protected), "protected_process_exited_during_upgrade")
            report.update(state="verified", exact_12_business_tables_preserved=True, exact_all_data_files_preserved=True, exact_credential_material_preserved=True, retained_before_sha256=digest(WORK / "before.json"), retained_after_sha256=digest(WORK / "after.json"), protected_outside_scope_count=len(protected), protected_observed_processes_unchanged=len(protected), historical_pid_observations=process["historical_pids"], integrity_check="ok", foreign_key_errors=0, business_counts={name: row["count"] for name, row in after["business_tables"].items()})
        except BaseException as error:
            report.update(state="failed_manual_review_required", error_code=str(error) if isinstance(error, Refusal) else "safe_upgrade_failure_details_withheld", outcome_requires_manual_review=True)
            write_new(WORK / "receipt.json", report)
            write_new(PUBLIC_RECEIPT, report)
            raise
        write_new(WORK / "receipt.json", report)
        write_new(PUBLIC_RECEIPT, report)
        return {"state": "verified", "receipt": str(PUBLIC_RECEIPT), "receipt_sha256": digest(PUBLIC_RECEIPT), "installed": report["installed"], "installer_execution": report["installer_execution"], "protected_observed_count": len(protected), "application_started": False, "manual_gui_verified": False, "paid_model_calls": 0}
    finally:
        if lock is not None:
            k.CloseHandle(lock)
        for handle in protected:
            k.CloseHandle(handle)


def main():
    sys.stdout.reconfigure(encoding="utf-8")
    sys.stderr.reconfigure(encoding="utf-8")
    parser = argparse.ArgumentParser(description=__doc__, allow_abbrev=False)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument("--survey-readonly", action="store_true")
    mode.add_argument("--execute-reviewed-upgrade", action="store_true")
    parser.add_argument("--reviewed-script-sha256")
    parser.add_argument("--readonly-survey-sha256")
    tokens = sys.argv[1:]
    flags = [token for token in tokens if token.startswith("--")]
    need(len(flags) == len(set(flags)) and not any("=" in flag for flag in flags), "duplicate_or_equals_flags_refused")
    args = parser.parse_args()
    hashes = (args.reviewed_script_sha256, args.readonly_survey_sha256)
    if args.execute_reviewed_upgrade:
        need(all(value and SHA.fullmatch(value) for value in hashes), "reviewed_script_and_survey_hashes_required")
        value = execute_upgrade(args)
    else:
        need(all(value is None for value in hashes), "hash_arguments_require_explicit_execution")
        value = survey_readonly() if args.survey_readonly else {"state": "review_only", "script_sha256": digest(Path(__file__)), "installer_sha256": INSTALLER_SHA, "source_freeze_sha256": FREEZE_SHA, "fixed_installation_directory": str(INSTALL), "fixed_data_directory": str(DATA), "windows_child_processes_started": 0, "application_started": False, "installer_executed": False, "registry_writes": False, "data_changes": False, "automatic_retry_or_resume": False}
    print(json.dumps(value, ensure_ascii=False))


if __name__ == "__main__":
    try:
        main()
    except BaseException as error:
        if isinstance(error, SystemExit):
            raise
        print(json.dumps({"state": "refused_or_failed", "error_code": str(error) if isinstance(error, Refusal) else "safe_failure_details_withheld", "automatic_retry_or_resume": False, "paid_model_calls": 0}))
        sys.exit(1)
