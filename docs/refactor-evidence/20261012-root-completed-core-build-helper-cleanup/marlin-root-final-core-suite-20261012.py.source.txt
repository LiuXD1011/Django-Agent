from pathlib import Path
import datetime, hashlib, importlib.util, json, os, subprocess, time

ROOT = Path('/home/liuxuedeng/Django-Agent')
CORE = ROOT / 'packages/core'
NODE = Path('/home/liuxuedeng/.nvm/versions/node/v22.23.2/bin/node')
FREEZE = ROOT / 'docs/refactor-evidence/20261012-integrated-poll-clock-source-freeze.json'
FREEZE_SHA = '46a4ad77664f71d593f6f53a7504ed95e1651c8a806ec80fe5b16af66112007e'
OUT = ROOT / 'docs/refactor-evidence/20261012-final-core-suite'
sha = lambda path: hashlib.sha256(path.read_bytes()).hexdigest()
assert sha(FREEZE) == FREEZE_SHA
frozen = json.loads(FREEZE.read_text())
spec = importlib.util.spec_from_file_location('marlin_final_core_build_source', ROOT / 'scripts/build_windows.py')
build = importlib.util.module_from_spec(spec)
spec.loader.exec_module(build)
assert build.source_inputs(ROOT) == frozen['sourceInputs']
assert all(sha(ROOT / path) == value for group in ('additionalReviewedTestInputs', 'localCompiledOutputs') for path, value in frozen[group].items())
assert NODE.is_file()
files = sorted(path.relative_to(CORE).as_posix() for path in (CORE / 'test').glob('*.test.ts'))
assert files
command = [str(NODE), '--import', 'tsx', '--test', '--test-concurrency=2', *files]
environment = {key: os.environ[key] for key in ('PATH', 'HOME', 'TMPDIR', 'LANG', 'LC_ALL') if key in os.environ}
environment.update(MARLIN_TEST_PYTHON='/tmp/marlin-parser-isolation-20261010/bin/python', MARLIN_CORE_OWNER_COMPILED_TEST='1', RAGAS_DO_NOT_TRACK='true', HF_HUB_DISABLE_TELEMETRY='1', DO_NOT_TRACK='1')
OUT.mkdir(exist_ok=False)
value = {'state': 'running_full_core_suite', 'startedUtc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'sourceFreezeSHA': FREEZE_SHA, 'command': command, 'cwd': str(CORE), 'fileCount': len(files), 'testConcurrency': 2, 'compiledOwnerTestsEnabled': True, 'environmentKeys': sorted(environment), 'realUserDataUsed': False, 'realVendorConfigurationUsed': False, 'guiActions': 0, 'businessSourceModified': False}
(OUT / 'attempt.json').write_text(json.dumps(value, indent=2))
print(json.dumps({'state': value['state'], 'sourceFreezeSHA': FREEZE_SHA, 'fileCount': len(files), 'testConcurrency': 2}), flush=True)
started = time.monotonic()
with (OUT / 'core-suite.log').open('w') as stream:
    result = subprocess.run(command, cwd=CORE, env=environment, stdout=stream, stderr=subprocess.STDOUT, check=False)
value.update(exitCode=result.returncode, durationSeconds=time.monotonic()-started, completedUtc=datetime.datetime.now(datetime.timezone.utc).isoformat(), sourceInputsUnchanged=build.source_inputs(ROOT)==frozen['sourceInputs'], localCompiledOutputsUnchanged=all(sha(ROOT / path)==expected for path, expected in frozen['localCompiledOutputs'].items()), logSHA=sha(OUT / 'core-suite.log'), logBytes=(OUT / 'core-suite.log').stat().st_size)
(OUT / 'execution.json').write_text(json.dumps(value, indent=2))
print(json.dumps({key: value[key] for key in ('exitCode', 'durationSeconds', 'sourceInputsUnchanged', 'localCompiledOutputsUnchanged', 'logSHA')}), flush=True)
raise SystemExit(result.returncode)
