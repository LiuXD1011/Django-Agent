import datetime, hashlib, importlib.util, json, sys, time
from pathlib import Path

ROOT = Path(r'\\wsl.localhost\Ubuntu-20.04\home\liuxuedeng\Django-Agent')
EVIDENCE = ROOT / 'docs/refactor-evidence/20261012-integrated-chinese-query-windows-build'
WORK = Path(r'C:\Users\lxd\AppData\Local\Temp\Marlin TS Chinese Query Final Build 20261012')
OUTPUT = Path(r'C:\Users\lxd\AppData\Local\Temp\Marlin TS Readiness Generation Installers 20261012')
NODE = Path(r'C:\Users\lxd\AppData\Local\Temp\Marlin Root Build Host 20261010\node-v22.23.2-win-x64\node.exe')
CACHE = Path(r'C:\Users\lxd\AppData\Local\Temp\Marlin TS Final Build 20261010\node-cache')
FREEZE = ROOT / 'docs/refactor-evidence/20261012-integrated-chinese-query-source-freeze.json'
EXPECTED = '39feca8035ebdb2c00d0dba9d603ff51a435f9bc766c525fe5fe9f8f92da1c3f'

def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

assert sys.platform == 'win32'
phase = sys.argv[1]
assert phase in ('prepare', 'package')
assert sha(FREEZE) == EXPECTED
build = load('marlin_chinese_query_build', ROOT / 'scripts/build_windows.py')
job = load('marlin_chinese_query_job', ROOT / 'scripts/desktop/windows-job.py')
frozen = json.loads(FREEZE.read_text(encoding='utf8'))
assert build.source_inputs(ROOT) == frozen['sourceInputs']
assert all(sha(ROOT / name) == value for name, value in frozen['additionalReviewedTestInputs'].items())
assert sha(NODE) == build.NODE_EXE_SHA256
assert sha(CACHE / f'node-v{build.NODE_VERSION}-win-x64.zip') == build.NODE_SHA256
if phase == 'prepare':
    assert not WORK.exists() and not OUTPUT.exists(), 'Fresh build and output directories required'
else:
    assert (WORK / '.marlin-build').read_text(encoding='utf8') == build.MARKER_VALUE
EVIDENCE.mkdir(exist_ok=True)
log, attempt, receipt = (EVIDENCE / (phase + suffix) for suffix in ('.log', '-attempt.json', '-execution.json'))
assert not any(path.exists() for path in (log, attempt, receipt))
if phase == 'prepare':
    WORK.mkdir()
environment = build.build_environment(NODE, WORK)
environment.update(PYTHONUTF8='1', PYTHONIOENCODING='utf-8')
command = [str(Path(sys.executable)), '-B', str(ROOT / 'scripts/build_windows.py'), '--prepare-only' if phase == 'prepare' else '--package-only', '--work-dir', str(WORK), '--build-node', str(NODE), '--node-cache', str(CACHE), '--output-dir', str(OUTPUT)]
value = {'phase': phase, 'startedUtc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'command': command, 'cwd': str(WORK), 'environmentKeys': sorted(environment), 'ownedJobTimeoutSeconds': 3600, 'sourceFreezeSHA': EXPECTED, 'sourceInputCount': len(frozen['sourceInputs']), 'additionalTestCount': len(frozen['additionalReviewedTestInputs']), 'installerExecuted': False, 'manualGUI': False, 'cleanBuild': phase == 'prepare', 'priorFailedPayloadReused': False}
with attempt.open('x', encoding='utf8') as stream:
    json.dump(value, stream, indent=2)
print(json.dumps(value), flush=True)
start = time.monotonic()
try:
    result = job.run_owned(command, WORK, environment, 3600, log)
    value.update(exitCode=result.returncode, durationSeconds=time.monotonic() - start, completedUtc=datetime.datetime.now(datetime.timezone.utc).isoformat(), log={'path': str(log), 'sha256': sha(log), 'bytes': log.stat().st_size}, ownedProcessTreeClosed=True)
except BaseException as error:
    value.update(exitCode=None, durationSeconds=time.monotonic() - start, completedUtc=datetime.datetime.now(datetime.timezone.utc).isoformat(), error=str(error), log={'path': str(log), 'sha256': sha(log) if log.exists() else None})
    with receipt.open('x', encoding='utf8') as stream:
        json.dump(value, stream, indent=2)
    raise
with receipt.open('x', encoding='utf8') as stream:
    json.dump(value, stream, indent=2)
print(json.dumps(value), flush=True)
raise SystemExit(result.returncode)
