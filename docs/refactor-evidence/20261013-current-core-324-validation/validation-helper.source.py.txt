def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def write(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def files_manifest(directories, extra=()):
    files = []
    for directory in directories:
        files.extend(p for p in (ROOT / directory).rglob('*') if p.is_file() and '__pycache__' not in p.parts and p.suffix != '.pyc')
    files.extend(ROOT / p for p in extra)
    return {str(p.relative_to(ROOT)): digest(p) for p in sorted(set(files)) if p.exists()}

def proc_table():
    answer = {}
    for path in Path('/proc').iterdir():
        if not path.name.isdecimal():
            continue
        try:
            raw = (path / 'stat').read_text(); tail = raw[raw.rfind(')') + 2:].split()
            answer[int(path.name)] = {'pid': int(path.name), 'ppid': int(tail[1]), 'state': tail[0], 'start_ticks': tail[19]}
        except (OSError, ValueError, IndexError):
            continue
    return answer

def owned_observe(root_pid, observed):
    table = proc_table(); owners = {root_pid}
    while True:
        new = {pid for pid, value in table.items() if value['ppid'] in owners}
        if new <= owners:
            break
        owners.update(new)
    for pid in owners:
        if pid not in table:
            continue
        value = table[pid]; key = (pid, value['start_ticks'])
        if key not in observed:
            value = dict(value)
            try:
                value['exe'] = str((Path('/proc') / str(pid) / 'exe').resolve(strict=True))
                value['argv'] = (Path('/proc') / str(pid) / 'cmdline').read_bytes().decode(errors='replace').split('\0')[:-1]
            except OSError:
                value['details'] = 'already terminal during observation'
            observed[key] = value

def execute(name, command, env):
    started = time.time(); observed = {}
    with (OUT / (name + '.stdout.log')).open('wb') as stdout, (OUT / (name + '.stderr.log')).open('wb') as stderr:
        child = subprocess.Popen(command, cwd=CORE, env=env, stdout=stdout, stderr=stderr)
        write(name + '.started.json', {'pid': child.pid, 'command': command, 'started_utc': datetime.datetime.fromtimestamp(started, datetime.timezone.utc).isoformat()})
        while child.poll() is None:
            owned_observe(child.pid, observed)
            time.sleep(0.15)
        code = child.wait()
    table = proc_table()
    survivors = [value for (pid, ticks), value in observed.items() if pid in table and table[pid]['start_ticks'] == ticks and table[pid]['state'] != 'Z']
    zombies = [value for (pid, ticks), value in observed.items() if pid in table and table[pid]['start_ticks'] == ticks and table[pid]['state'] == 'Z']
    record = {'command': command, 'cwd': str(CORE), 'started_utc': datetime.datetime.fromtimestamp(started, datetime.timezone.utc).isoformat(), 'elapsed_seconds': time.time()-started, 'pid': child.pid, 'exit_code': code, 'stdout': name+'.stdout.log', 'stderr': name+'.stderr.log'}
    write(name + '.result.json', record)
    write(name + '.owned-process-observation.json', {'method': 'strict observed descendant ancestry, PID plus /proc start_ticks; no service/port/name matching and no signal sent', 'observed': list(observed.values()), 'survivors_after_terminal': survivors, 'zombies_after_terminal': zombies})
    print(json.dumps(record, ensure_ascii=False), flush=True)
    if code != 0 or survivors:
        raise RuntimeError(name + ' failed or observed owned processes remain; no rerun or kill')
    return record
