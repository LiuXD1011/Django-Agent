def utc():
 return datetime.datetime.now(datetime.timezone.utc).isoformat()

def binding(path):
 data = (ROOT / path).read_bytes()
 return {'path': path, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()}

def save(name, data):
 (OUT / name).write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')

def observation():
 return {'utc': utc(), 'head': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
 'logical_stage_sha256': hashlib.sha256(subprocess.check_output(['git', 'ls-files', '--stage'], cwd=ROOT)).hexdigest(),
 'git_write_operations_by_this_worker': False}

def ids(text, ui=False):
 return re.findall(r'^\| (UI-[A-Z]+-\d{3}|(?:[A-JMR]-\d{2}|E-GOLD|PERF)) \|', text, re.M) if not ui else re.findall(r'^\| (UI-[A-Z]+-\d{3}) \|', text, re.M)

  now = (ROOT / path).read_text()
  assert hashlib.sha256(pre.encode()).hexdigest() == next(x['sha256'] for x in before['documents'] if x['path'] == path)
  patch = ''.join(difflib.unified_diff(pre.splitlines(True), now.splitlines(True), fromfile='a/' + path, tofile='b/' + path))
  assert patch
  diff_parts.append(patch)
  new_lines = [line[1:] for line in patch.splitlines() if line.startswith('+') and not line.startswith('+++')]
  changes.append({'path': path, 'added_lines': len(new_lines), 'removed_lines': sum(line.startswith('-') and not line.startswith('---') for line in patch.splitlines())})
  for line in new_lines:
   if line.rstrip() != line:
    failures.append('new trailing whitespace: ' + path)
   links.extend((path, target) for target in re.findall(r'\[[^\]]*\]\(([^)]+)\)', line))
  if Path(path).name == 'TEST-MATRIX.md':
   old_all, new_all = ids(pre), ids(now)
   old_ui, new_ui = ids(pre, True), ids(now, True)

   new_groups = [x for x in new_all if not x.startswith('UI-')]
