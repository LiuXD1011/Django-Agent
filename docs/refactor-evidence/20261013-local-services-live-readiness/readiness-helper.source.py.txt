def digest(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def existence(path):

def command(arguments, env, mode):
    started=time.monotonic()
    try:
        r=subprocess.run(arguments,env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=20)
        stdout=r.stdout.decode(errors='replace').strip()
        safe=None
        if mode == 'version':
            found=re.search(r'(?:^|[^0-9])(v?\d+\.\d+\.\d+(?:[-+][0-9A-Za-z.-]+)?)',stdout)
            safe=found.group(1) if found else None
        elif mode == 'json':
            try:
                value=json.loads(stdout);safe={key:value[key] for key in ['ServerVersion','OperatingSystem','OSType','Architecture','ContainersRunning'] if key in value}
            except (ValueError,TypeError): pass
        errors=r.stderr.decode(errors='replace').lower()
        error_class='permission_denied' if 'permission denied' in errors else 'daemon_unreachable' if any(s in errors for s in ['cannot connect','connection refused','connectex','pipe','is the docker daemon running']) else 'nonzero_exit' if r.returncode else None
        return {'argv':arguments,'exit_code':r.returncode,'duration_seconds':time.monotonic()-started,'safe_output':safe,'error_class':error_class,'raw_output_saved':False}
    except subprocess.TimeoutExpired:
        return {'argv':arguments,'exit_code':None,'duration_seconds':time.monotonic()-started,'error_class':'probe_timeout','raw_output_saved':False}
    except OSError as e:
        return {'argv':arguments,'exit_code':None,'duration_seconds':time.monotonic()-started,'error_class':type(e).__name__,'raw_output_saved':False}
def probe(host):
    started=time.monotonic(); result={'host':host,'port':3000,'path':'/api/public/health','request':'GET without auth/cookie/proxy','body_or_headers_saved':False}
    connection=http.client.HTTPConnection(host,3000,timeout=3)
    try:
        connection.request('GET','/api/public/health');response=connection.getresponse();result.update(status=response.status,healthy=response.status==200)
    except Exception as e: result.update(status=None,healthy=False,error_class=type(e).__name__)
    finally:connection.close()
    result['duration_seconds']=time.monotonic()-started
    return result
