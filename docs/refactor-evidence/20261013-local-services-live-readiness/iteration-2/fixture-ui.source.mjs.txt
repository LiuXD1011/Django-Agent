import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { dirname, join } from 'node:path';
const fixture = dirname(fileURLToPath(import.meta.url));
const config = JSON.parse(readFileSync(join(fixture, 'fixture.json'), 'utf8'));
const { createServer } = await import(config.root + '/frontend/node_modules/vite/dist/node/index.js');
const { default: vue } = await import(config.root + '/frontend/node_modules/@vitejs/plugin-vue/dist/index.mjs');
const origin = `http://127.0.0.1:${config.core_port}`;
const server = await createServer({ root: config.root + '/frontend', configFile: false, envDir: false,
  plugins: [vue()], cacheDir: join(fixture, 'cache/vite'), clearScreen: false,
  server: { host: '127.0.0.1', port: config.ui_port, strictPort: true, open: false,
    proxy: { '/api': origin, '/files': origin, '/health': origin } } });
let closing;
const stop = () => closing ??= server.close().then(() => { process.exitCode = 0; });
process.once('SIGINT', stop); process.once('SIGTERM', stop);
await server.listen(); console.log(`Fixture UI listening at http://127.0.0.1:${config.ui_port}`);
