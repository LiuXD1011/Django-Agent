def group_rows(text):
    result={}
    for line in text.splitlines():
        m=re.match(r'^\| ([A-Z]-\d{2}|E-GOLD|PERF) \|',line)
        if m:
            cells=[c.strip() for c in line.strip('|').split('|')]
            assert len(cells)==4,(m.group(1),len(cells))
            result[m.group(1)]=re.match(r'(通过|未执行|未完成|环境失败|产品失败|缺少配置)',cells[3]).group(1)
    return result
