def cli(name, arguments, expected_exit, expected_state=None, expected_error=None):
    result = subprocess.run([sys.executable, '-B', str(HARNESS), *arguments], capture_output=True, text=True, timeout=10)
    assert result.returncode == expected_exit, (name, result.returncode)
    value = None
    if expected_state or expected_error:
        value = json.loads(result.stdout)
        if expected_state: assert value['state'] == expected_state
        if expected_error: assert value['error_code'] == expected_error
    if expected_state == 'review_only':
        assert value['windows_process_operations'] == 0
        assert value['application_started'] is False and value['installer_executed'] is False
    checks.append({'name': name, 'exit_code': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr, 'passed': True})

def request_case(name, value, expected_error=None, supplied_sha=None):
    with tempfile.TemporaryDirectory(prefix='marlin-integrated-binding-guard-') as directory:
        harness.EVIDENCE = Path(directory)
        path = harness.EVIDENCE / 'reviewed-binding-request-1.json'
        path.write_text(json.dumps(value), encoding='utf8')
        args = SimpleNamespace(iteration=1, binding_request_sha256=supplied_sha or harness.digest(path), source_freeze_sha256='c'*64)
        try:
            result = harness.request(args, legacy)
        except harness.Refusal as error:
            assert str(error) == expected_error, (name, str(error), expected_error)
            checks.append({'name': name, 'error_code': str(error), 'passed': True, 'fixture_only': True})
        else:
            assert expected_error is None, name
            assert result['iteration'] == 1 and result['sourceFreezeSHA'] == 'c'*64
            checks.append({'name': name, 'passed': True, 'fixture_only': True})
