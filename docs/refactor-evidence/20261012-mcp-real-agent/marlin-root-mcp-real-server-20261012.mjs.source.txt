const sessionHash = value => createHash('sha256').update(String(value ?? '')).digest('hex');
const body = async request => {
  const chunks = []; let length = 0;
  for await (const chunk of request) {
    length += chunk.length;
    if (length > 16384) throw new Error('Bounded request exceeded');
    chunks.push(chunk);
  }
  return JSON.parse(Buffer.concat(chunks).toString('utf8'));
};
const http = createServer(async (request, response) => {
  try {
    if (request.url !== '/mcp') { response.writeHead(404).end(); return; }
    const matched = request.headers.authorization === syntheticAuthorization;
    if (!matched) {
      record({ event: 'authentication_rejected', matched: false, method: request.method });
      response.writeHead(401).end(); return;
    }
    const payload = request.method === 'POST' ? await body(request) : undefined;
    const method = ['initialize', 'notifications/initialized', 'tools/list', 'tools/call', 'ping'].includes(payload?.method) ? payload.method : null;
    const setup = request.headers['x-marlin-qa-phase'] === 'setup-smoke';
    record({ event: 'protocol_request', method, httpMethod: request.method, matched, sessionSHA: sessionHash(request.headers['mcp-session-id']), setupSmoke: setup });
    let session = sessions.get(String(request.headers['mcp-session-id'] ?? ''));
    if (!session && payload?.method === 'initialize') {
      const transport = new StreamableHTTPServerTransport({ sessionIdGenerator: randomUUID, enableJsonResponse: true });
      const server = new Server({ name: 'marlin-qa-purchase-calculator', version: '1.0.0' }, { capabilities: { tools: {} } });
      server.setRequestHandler(ListToolsRequestSchema, () => ({ tools: [{ name: 'qa_purchase_total', description: 'Compute an acceptance purchase subtotal and return the unique MCP service receipt. No file or network access.', inputSchema: { type: 'object', properties: { quantity: { type: 'integer', minimum: 0, maximum: 1000 }, unit_price: { type: 'integer', minimum: 0, maximum: 100000 } }, required: ['quantity', 'unit_price'], additionalProperties: false } }] }));
      server.setRequestHandler(CallToolRequestSchema, request => {
        const args = request.params.arguments ?? {};
        if (request.params.name !== 'qa_purchase_total' || Object.keys(args).sort().join(',') !== 'quantity,unit_price' || !Number.isInteger(args.quantity) || args.quantity < 0 || args.quantity > 1000 || !Number.isInteger(args.unit_price) || args.unit_price < 0 || args.unit_price > 100000) return { isError: true, content: [{ type: 'text', text: 'Invalid bounded purchase arguments' }] };
        const result = { quantity: args.quantity, unit_price: args.unit_price, subtotal: args.quantity * args.unit_price, receipt_id: `${instance}:${++calls}` };
        record({ event: 'tool_called', tool: 'qa_purchase_total', setupSmoke: setup, ...result });
        return { content: [{ type: 'text', text: JSON.stringify(result) }] };
      });
      await server.connect(transport);
      session = { server, transport };
    }
    if (!session) { response.writeHead(404).end(); return; }
    await session.transport.handleRequest(request, response, payload);
    if (session.transport.sessionId) sessions.set(session.transport.sessionId, session);
    if (request.method === 'DELETE') {
      sessions.delete(String(request.headers['mcp-session-id']));
      record({ event: 'session_terminated', setupSmoke: setup, sessionSHA: sessionHash(request.headers['mcp-session-id']) });
    }
  } catch {
    record({ event: 'protocol_handler_failed', rawDetailsRecorded: false });
    if (!response.headersSent) response.writeHead(400).end();
    else response.end();
  }
});

  process.exitCode = 0;
}
