FREEZE = ROOT / 'docs/refactor-evidence/20261012-integrated-poll-clock-source-freeze.json'

def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

assert sys.platform == 'win32'
phase = sys.argv[1]
assert phase in ('prepare', 'package')
assert sha(FREEZE) == EXPECTED
assert sha(ROOT / 'scripts/build_windows.py') == BUILD_SHA
assert sha(ROOT / 'scripts/desktop/windows-job.py') == JOB_SHA
build = load('marlin_poll_clock_build', ROOT / 'scripts/build_windows.py')
job = load('marlin_poll_clock_job', ROOT / 'scripts/desktop/windows-job.py')
frozen = json.loads(FREEZE.read_text(encoding='utf8'))
assert build.source_inputs(ROOT) == frozen['sourceInputs']
assert all(sha(ROOT / name) == value for group in ('additionalReviewedTestInputs', 'localCompiledOutputs') for name, value in frozen[group].items())
assert sha(NODE) == build.NODE_EXE_SHA256
assert sha(CACHE / f'node-v{build.NODE_VERSION}-win-x64.zip') == build.NODE_SHA256
assert (WORK / '.marlin-build').read_text(encoding='utf8') == build.MARKER_VALUE
if phase == 'prepare':
    assert sha(FAILED) == FAILED_SHA
    failed = json.loads(FAILED.read_text(encoding='utf8'))
    assert failed['exitCode'] == 1 and failed['ownedProcessTreeClosed'] is True
    assert failed['sourceFreezeSHA'] == frozen['priorSourceFreezeSHA']
    assert failed['cwd'] == str(WORK)
    assert sha(FAILED.parent / 'prepare.log') == failed['log']['sha256']
    assert not OUTPUT.exists()
    assert all(not (WORK / name).exists() for name in ('app', 'prepared-build.json', 'windows-gates.json', 'core-verification.json', 'python-verification.json'))
    assert sha(WORK / 'frontend/src/views/KnowledgeDetail.processing-records.test.mjs') == frozen['changesFromPrior']['frontend/src/views/KnowledgeDetail.processing-records.test.mjs']['before']
else:
    prepared = json.loads((EVIDENCE / 'prepare-execution.json').read_text(encoding='utf8'))

    assert prepared['ownedProcessTreeClosed'] is True
    assert sha(EVIDENCE / 'prepare.log') == prepared['log']['sha256']
EVIDENCE.mkdir(exist_ok=True)
log, attempt, receipt = (EVIDENCE / (phase + suffix) for suffix in ('.log', '-attempt.json', '-execution.json'))

environment = build.build_environment(NODE, WORK)
environment.update(PYTHONUTF8='1', PYTHONIOENCODING='utf-8')
command = [str(Path(sys.executable)), '-B', str(ROOT / 'scripts/build_windows.py'), '--prepare-only' if phase == 'prepare' else '--package-only', '--work-dir', str(WORK), '--build-node', str(NODE), '--node-cache', str(CACHE), '--output-dir', str(OUTPUT)]
value = {'phase': phase, 'startedUtc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'command': command, 'cwd': str(WORK), 'environmentKeys': sorted(environment), 'ownedJobTimeoutSeconds': 3600, 'sourceFreezeSHA': EXPECTED, 'sourceInputCount': len(frozen['sourceInputs']), 'additionalTestCount': len(frozen['additionalReviewedTestInputs']), 'installerExecuted': False, 'manualGUI': False, 'cleanBuild': phase == 'prepare', 'existingBuilderOwnedStage': True, 'builderResetsAllSourceAndDependencyStaging': phase == 'prepare', 'priorFailedPayloadReused': False, 'priorFailedPrepareReceiptSHA': FAILED_SHA}
with attempt.open('x', encoding='utf8') as stream:

    result = job.run_owned(command, WORK, environment, 3600, log)
    value.update(exitCode=result.returncode, durationSeconds=time.monotonic() - start, completedUtc=datetime.datetime.now(datetime.timezone.utc).isoformat(), log={'path': str(log), 'sha256': sha(log), 'bytes': log.stat().st_size}, ownedProcessTreeClosed=True)
except BaseException as error:

    with receipt.open('x', encoding='utf8') as stream:
        json.dump(value, stream, indent=2)
    raise
with receipt.open('x', encoding='utf8') as stream:
    json.dump(value, stream, indent=2)
print(json.dumps(value), flush=True)
raise SystemExit(result.returncode)
